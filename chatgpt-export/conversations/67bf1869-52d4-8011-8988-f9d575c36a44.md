### **1. Backend (Flask) - `app.py`**
Handles user authentication and serves the frontend.

```python
from flask import Flask, render_template, request, redirect, url_for, session
import database
import email_reminder

app = Flask(__name__)
app.secret_key = 'your_secret_key'

@app.route('/')
def home():
    return render_template('index.html')

@app.route('/login', methods=['GET', 'POST'])
def login():
    if request.method == 'POST':
        email = request.form['email']
        password = request.form['password']
        user = database.authenticate_user(email, password)
        if user:
            session['user'] = email
            return redirect(url_for('dashboard'))
        return "Login failed!"
    return render_template('login.html')

@app.route('/dashboard')
def dashboard():
    if 'user' in session:
        return render_template('dashboard.html', user=session['user'])
    return redirect(url_for('login'))

@app.route('/logout')
def logout():
    session.pop('user', None)
    return redirect(url_for('home'))

if __name__ == '__main__':
    app.run(debug=True)
```

### **2. User Data Handling - `database.py`**
Stores user information in an Excel file.

```python
import pandas as pd
import bcrypt

FILE_PATH = "users.xlsx"

def load_users():
    try:
        return pd.read_excel(FILE_PATH)
    except FileNotFoundError:
        return pd.DataFrame(columns=['Email', 'Password'])

def save_users(df):
    df.to_excel(FILE_PATH, index=False)

def register_user(email, password):
    df = load_users()
    hashed_password = bcrypt.hashpw(password.encode(), bcrypt.gensalt()).decode()
    df = df.append({'Email': email, 'Password': hashed_password}, ignore_index=True)
    save_users(df)

def authenticate_user(email, password):
    df = load_users()
    user = df[df['Email'] == email]
    if not user.empty:
        stored_password = user.iloc[0]['Password']
        return bcrypt.checkpw(password.encode(), stored_password.encode())
    return False
```

### **3. Automated Email Reminders - `email_reminder.py`**
Sends email reminders using SMTP.

```python
import smtplib
from email.mime.text import MIMEText
from datetime import datetime, timedelta

SMTP_SERVER = 'smtp.gmail.com'
SMTP_PORT = 587
SENDER_EMAIL = 'your_email@gmail.com'
SENDER_PASSWORD = 'your_password'

def send_email(recipient, subject, body):
    msg = MIMEText(body)
    msg['Subject'] = subject
    msg['From'] = SENDER_EMAIL
    msg['To'] = recipient

    with smtplib.SMTP(SMTP_SERVER, SMTP_PORT) as server:
        server.starttls()
        server.login(SENDER_EMAIL, SENDER_PASSWORD)
        server.sendmail(SENDER_EMAIL, recipient, msg.as_string())

def send_appointment_reminder(email, appointment_date):
    reminder_date = appointment_date - timedelta(days=1)
    if datetime.now().date() == reminder_date.date():
        send_email(email, "Appointment Reminder", "Your appointment is scheduled for tomorrow.")
```

import csv

DATABASE_FILE = "backend/users.csv"

class MyHandler(http.server.SimpleHTTPRequestHandler):
    def do_POST(self):
        if self.path == "/register":
            content_length = int(self.headers['Content-Length'])
            post_data = self.rfile.read(content_length).decode('utf-8')
            data = urllib.parse.parse_qs(post_data)

            self.end_headers()
            self.wfile.write(b"Not found")

## **✅ Step 3: Web Server (`server.py`)**
Handles **user login, registration**, and serves static pages.

```python
import http.server
import socketserver
import urllib.parse
import database
import email_service

PORT = 8000

class MyHandler(http.server.SimpleHTTPRequestHandler):
    def do_POST(self):
        content_length = int(self.headers['Content-Length'])
        post_data = self.rfile.read(content_length).decode('utf-8')
        data = urllib.parse.parse_qs(post_data)

        if self.path == "/register":
            username = data.get("username", [""])[0]
            email = data.get("email", [""])[0]

            if username and email and not database.user_exists(username):
                database.save_user(username, email)
                email_service.send_welcome_email(email)
                self._send_response(200, "Registration successful! <a href='/'>Go back</a>")
            else:
                self._send_response(400, "User already exists or invalid data.")

        elif self.path == "/login":
            username = data.get("username", [""])[0]

            if database.user_exists(username):
                email = [user[1] for user in database.load_users() if user[0] == username][0]
                email_service.send_login_email(email)
                self._send_response(200, f"Welcome back, {username}! <a href='/dashboard.html'>Go to Dashboard</a>")
            else:
                self._send_response(400, "User not found. Please register.")

        else:
            self._send_response(404, "Not found")

    def _send_response(self, code, message):
        self.send_response(code)
        self.send_header("Content-type", "text/html")
        self.end_headers()
        self.wfile.write(message.encode())

# Start the server
with socketserver.TCPServer(("", PORT), MyHandler) as httpd:
    print(f"Server running at http://127.0.0.1:{PORT}")
    httpd.serve_forever()
```

    password = "your_email_password"

    msg['From'] = from_email

import http.server
import socketserver
import os

PORT = 8000
PROJECT_ROOT = "/storage/emulated/0/ai_appointment_system/frontend"

class MyHandler(http.server.SimpleHTTPRequestHandler):
    def do_GET(self):
        """Serve static files correctly."""
        if self.path.startswith("/static/"):
            self.serve_static_file()
        else:
            if self.path == "/":
                self.path = "/index.html"
            return http.server.SimpleHTTPRequestHandler.do_GET(self)

    def serve_static_file(self):
        """Serve static files properly."""
        static_file_path = os.path.join(STATIC_DIR, self.path.lstrip("/static/"))
        if os.path.exists(static_file_path):
            try:
                self.send_response(200)
                if self.path.endswith(".css"):
                    self.send_header("Content-Type", "text/css")
                elif self.path.endswith(".mp4"):
                    self.send_header("Content-Type", "video/mp4")
                    self.send_header("Accept-Ranges", "bytes")
                self.end_headers()

                with open(static_file_path, "rb") as file:
                    while True:
                        chunk = file.read(1024 * 64)  # Read 64 KB chunks
                        if not chunk:
                            break
                        self.wfile.write(chunk)
            except (BrokenPipeError, ConnectionResetError):
                print("Client closed connection (Broken Pipe).")
        else:
            self.send_error(404, "Static file not found")

os.chdir(PROJECT_ROOT)

with socketserver.TCPServer(("", PORT), MyHandler) as httpd:
    print(f"Server running at http://127.0.0.1:{PORT}")
    try:
        httpd.serve_forever()
    except Exception as e:
        print(f"Error running the server: {e}")
