    ws_main.cell(row=row, column=4, value=data["social"])
    ws_main.cell(row=row, column=5, value=f"{company}_Products")

# Product sheets
for company, data in companies.items():
    ws = wb.create_sheet(title=f"{company}_Products")

    headers = [
        "Product Name",
        "Retail Price",
        "Estimated Margin %",
        "Estimated Cost",
        "Department",
        "Stock Status",
        "Category"
    ]

    for col, header in enumerate(headers, 1):
        cell = ws.cell(row=1, column=col, value=header)
        cell.font = Font(bold=True)

    # Dropdown validation
    status_validation = DataValidation(
        type="list",
        formula1='"In Stock,Low Stock,Out of Stock"',
        allow_blank=True
    )
    category_validation = DataValidation(
        type="list",
        formula1='"Electronics,Apparel,Accessories,Industrial"',
        allow_blank=True
    )

    ws.add_data_validation(status_validation)
    ws.add_data_validation(category_validation)

    r = 2

        ws.cell(row=r, column=4, value=cost)
        ws.cell(row=r, column=5, value="Sales Division")
        ws.cell(row=r, column=6, value="In Stock")
        ws.cell(row=r, column=7, value=data["industry"])

        category_validation.add(ws.cell(row=r, column=7))

        r += 1
