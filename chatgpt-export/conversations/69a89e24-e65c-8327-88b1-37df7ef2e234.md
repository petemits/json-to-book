const position = geometry.attributes.position;

# Example: Morphing Creatures in the Browser

Save this as **`morph.html`** and open it in your browser.

```html
<!DOCTYPE html>
<html>
<head>
<title>Mesh Morph Creatures</title>
<style>
body { margin:0; overflow:hidden; background:black;}
</style>
</head>

<body>

<script src="https://cdn.jsdelivr.net/npm/three@0.158.0/build/three.min.js"></script>

<script>

const scene = new THREE.Scene();

const camera = new THREE.PerspectiveCamera(
75,
window.innerWidth/window.innerHeight,
0.1,
1000
);

camera.position.z = 60;

const renderer = new THREE.WebGLRenderer();
renderer.setSize(window.innerWidth, window.innerHeight);
document.body.appendChild(renderer.domElement);

const particleCount = 2000;

const geometry = new THREE.BufferGeometry();

const positions = new Float32Array(particleCount * 3);

for(let i=0;i<particleCount;i++){

positions[i*3] = (Math.random()-0.5)*50;
positions[i*3+1] = (Math.random()-0.5)*50;
positions[i*3+2] = (Math.random()-0.5)*50;

}

geometry.setAttribute(
'position',
new THREE.BufferAttribute(positions,3)
);

const material = new THREE.PointsMaterial({
color:0x00ffff,
size:0.8
});

const points = new THREE.Points(geometry, material);

scene.add(points);


function generateShape(type){

let target=[];

for(let i=0;i<particleCount;i++){

let x,y,z;

if(type=="bird"){

x=(Math.random()-0.5)*40;
y=Math.sin(x*0.2)*10;
z=(Math.random()-0.5)*5;

}

else if(type=="fish"){

x=Math.cos(i)*15;
y=Math.sin(i)*5;
z=(Math.random()-0.5)*5;

}

else if(type=="human"){

x=(Math.random()-0.5)*10;
y=(Math.random()*30)-15;
z=(Math.random()-0.5)*5;

}

else if(type=="butterfly"){

x=Math.sin(i)*20;
y=Math.cos(i*2)*10;
z=(Math.random()-0.5)*5;

}

target.push(x,y,z);

}

return target;

}

let shapes=[
generateShape("bird"),
generateShape("fish"),
generateShape("human"),
generateShape("butterfly")
];

let currentShape=0;

function morph(){

let pos=geometry.attributes.position.array;
let target=shapes[currentShape];

for(let i=0;i<particleCount*3;i++){

pos[i]+= (target[i]-pos[i])*0.02;

}

geometry.attributes.position.needsUpdate=true;

}

setInterval(()=>{

currentShape++;
if(currentShape>=shapes.length) currentShape=0;

},6000);


function animate(){

requestAnimationFrame(animate);

morph();

renderer.render(scene,camera);

}

animate();

</script>

</body>
</html>
```
