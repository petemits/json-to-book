## 1. 🔍 SCRAPER (Website → Data Extraction)

```javascript
function scrapeWebsite(url) {
  try {
    var html = UrlFetchApp.fetch(url, {muteHttpExceptions: true}).getContentText();

    var email = html.match(/[A-Z0-9._%+-]+@[A-Z0-9.-]+\.[A-Z]{2,}/i);
    var description = html.match(/<meta name="description" content="(.*?)"/i);
    var image = html.match(/<meta property="og:image" content="(.*?)"/i);

    var socials = [];
    if (html.includes("instagram.com")) socials.push("Instagram");
    if (html.includes("facebook.com")) socials.push("Facebook");
    if (html.includes("linkedin.com")) socials.push("LinkedIn");

    return {
      email: email ? email[0] : "",
      description: description ? description[1] : "",
      image: image ? image[1] : "",
      socials: socials.join(", ")
    };

  } catch (e) {
    return {email:"", description:"", image:"", socials:""};
  }
}
```

function scrapeWebsite(url) {
  try {
    var res = UrlFetchApp.fetch(url, {muteHttpExceptions:true});
    var html = res.getContentText();

    var email = (html.match(/[A-Z0-9._%+-]+@[A-Z0-9.-]+\.[A-Z]{2,}/i) || [""])[0];
    var desc = (html.match(/<meta name="description" content="(.*?)"/i) || ["",""])[1];
    var image = (html.match(/<meta property="og:image" content="(.*?)"/i) || ["",""])[1];

    var socials = [];
    if (html.includes("instagram.com")) socials.push("Instagram");
    if (html.includes("facebook.com")) socials.push("Facebook");
    if (html.includes("linkedin.com")) socials.push("LinkedIn");

    return {
      email: email,
      description: desc,
      image: image,
      socials: socials.join(", ")
    };

  } catch(e) {
    return {email:"", description:"", image:"", socials:""};
  }
}

function isDuplicate(email) {
  var sheets = ["RealEstate","Restaurants","Contractors"];
  var ss = SpreadsheetApp.getActive();

  for (var s of sheets) {
    var data = ss.getSheetByName(s).getDataRange().getValues();
    for (var i=1;i<data.length;i++) {
      if (data[i][3] === email && email !== "") return true;
    }
