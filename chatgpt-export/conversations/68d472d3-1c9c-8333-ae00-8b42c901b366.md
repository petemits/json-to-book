```python
import pyaudio
import numpy as np

# ----------------------------
# Configuration
# ----------------------------
CHUNK = 1024           # Number of audio samples per buffer
FORMAT = pyaudio.paInt16
CHANNELS = 1
RATE = 44100           # Samples per second
MUTE_LEVEL = 1000      # Amplitude limit for "steady-muted" effect

# ----------------------------
# DSP Function: Steady Muted
# ----------------------------
def steady_muted(signal, level=MUTE_LEVEL):
    # Limit the peaks of the waveform to reduce sudden loud sounds
    return np.clip(signal, -level, level)

# ----------------------------
# Initialize PyAudio
# ----------------------------
p = pyaudio.PyAudio()

# Input stream (microphone)
stream_in = p.open(format=FORMAT,
                   channels=CHANNELS,
                   rate=RATE,
                   input=True,
                   frames_per_buffer=CHUNK)

# Output stream (headphones)
stream_out = p.open(format=FORMAT,
                    channels=CHANNELS,
                    rate=RATE,
                    output=True,
                    frames_per_buffer=CHUNK)

print("Real-time DSP running. Speak into the microphone!")

# ----------------------------
# Main Loop
# ----------------------------
try:
    while True:
        # Read audio from mic
        data = stream_in.read(CHUNK, exception_on_overflow=False)
        audio = np.frombuffer(data, dtype=np.int16)
        
        # Apply DSP
        processed = steady_muted(audio)
        
        # Output to headphones
        stream_out.write(processed.tobytes())
except KeyboardInterrupt:
    print("Stopping...")
finally:
    stream_in.stop_stream()
    stream_in.close()
    stream_out.stop_stream()
    stream_out.close()
    p.terminate()
```

## **4. Example: Simple Steady-Muted DSP in Python**

```python
import pyaudio
import numpy as np

# Configuration
CHUNK = 1024       # Number of audio samples per buffer
RATE = 44100       # Sample rate (Hz)
FORMAT = pyaudio.paInt16
CHANNELS = 1
MUTE_LEVEL = 1000  # Max amplitude for muted effect

# DSP function
def steady_muted(signal, level=MUTE_LEVEL):
    # Limit amplitude to create muted/steady sound
    return np.clip(signal, -level, level)

# Initialize PyAudio
p = pyaudio.PyAudio()

# Input stream
stream_in = p.open(format=FORMAT, channels=CHANNELS, rate=RATE, input=True, frames_per_buffer=CHUNK)

# Output stream
stream_out = p.open(format=FORMAT, channels=CHANNELS, rate=RATE, output=True, frames_per_buffer=CHUNK)

print("Real-time DSP running. Speak into the mic...")

try:
    while True:
        # Capture audio
        data = stream_in.read(CHUNK, exception_on_overflow=False)
        audio = np.frombuffer(data, dtype=np.int16)
        
        # Apply DSP
        processed = steady_muted(audio)
        
        # Output processed audio
        stream_out.write(processed.tobytes())
except KeyboardInterrupt:
    print("Stopped.")
finally:
    stream_in.stop_stream()
    stream_in.close()
    stream_out.stop_stream()
    stream_out.close()
    p.terminate()
```
