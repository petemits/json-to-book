### **4. Automated Email & Inventory Alerts**
#### **File: `scheduler.py`**
```python
import smtplib
import pandas as pd
import schedule
import time

INVENTORY_FILE = "inventory.xlsx"
USER_FILE = "users.xlsx"

EMAIL = "your-email@gmail.com"
PASSWORD = "your-app-password"

# Send Email
def send_email(to_email, subject, message):
    server = smtplib.SMTP("smtp.gmail.com", 587)
    server.starttls()
    server.login(EMAIL, PASSWORD)
    
    email_message = f"Subject: {subject}\n\n{message}"
    server.sendmail(EMAIL, to_email, email_message)
    server.quit()

# Check inventory & send alerts
def check_inventory():
    df = pd.read_excel(INVENTORY_FILE)
    for _, row in df.iterrows():
        if row["Stock"] < 5:
            send_email(EMAIL, "Low Inventory Alert", f"{row['Product']} is running low!")

# Daily Newsletter
def send_newsletter():
    users = pd.read_excel(USER_FILE)["Email"].values
    for email in users:
        send_email(email, "Daily Newsletter", "Hello! Here's your daily newsletter.")

# Schedule tasks
schedule.every().day.at("09:00").do(send_newsletter)
schedule.every().hour.do(check_inventory)

def run_scheduler():
    while True:
        schedule.run_pending()
        time.sleep(60)

# Run in a separate thread
if __name__ == "__main__":
    run_scheduler()
```

## **1. `app.py` (Flask Website + CSV for Data)**
```python
from flask import Flask, render_template, request, redirect, url_for, flash, session
import csv
import os
import smtplib
import time
import threading
import schedule

# Initialize Flask app
app = Flask(__name__)
app.secret_key = "secret123"

USER_FILE = "users.csv"
INVENTORY_FILE = "inventory.csv"

EMAIL = "your-email@gmail.com"  # Update with real email
PASSWORD = "your-app-password"  # Update with real password

# Ensure necessary files exist
if not os.path.exists(USER_FILE):
    with open(USER_FILE, "w", newline="") as file:
        writer = csv.writer(file)
        writer.writerow(["Email", "Password"])

if not os.path.exists(INVENTORY_FILE):
    with open(INVENTORY_FILE, "w", newline="") as file:
        writer = csv.writer(file)
        writer.writerow(["Product", "Stock"])

# Homepage
@app.route('/')
def home():
    return render_template('index.html')

# User Registration
@app.route('/register', methods=['GET', 'POST'])
def register():
    if request.method == 'POST':
        email = request.form['email']
        password = request.form['password']
        
        with open(USER_FILE, "r") as file:
            reader = csv.reader(file)
            users = list(reader)
        
        if any(row[0] == email for row in users):
            flash("User already exists!", "danger")
        else:
            with open(USER_FILE, "a", newline="") as file:
                writer = csv.writer(file)
                writer.writerow([email, password])
            flash("Registration successful!", "success")
            return redirect(url_for('login'))

    return render_template('register.html')

# User Login
@app.route('/login', methods=['GET', 'POST'])
def login():
    if request.method == 'POST':
        email = request.form['email']
        password = request.form['password']
        
        with open(USER_FILE, "r") as file:
            reader = csv.reader(file)
            users = list(reader)
        
        if any(row[0] == email and row[1] == password for row in users):
            session['user'] = email
            return redirect(url_for('dashboard'))
        else:
            flash("Invalid credentials!", "danger")

    return render_template('login.html')

# Dashboard - Inventory Management
@app.route('/dashboard', methods=['GET', 'POST'])
def dashboard():
    if 'user' not in session:
        return redirect(url_for('login'))

    inventory = []
    with open(INVENTORY_FILE, "r") as file:
        reader = csv.reader(file)
        inventory = list(reader)[1:]

    if request.method == 'POST':
        product = request.form['product']
        stock = int(request.form['stock'])
        
        found = False
        for row in inventory:
            if row[0] == product:
                row[1] = str(int(row[1]) + stock)
                found = True
        
        if not found:
            inventory.append([product, stock])
        
        with open(INVENTORY_FILE, "w", newline="") as file:
            writer = csv.writer(file)
            writer.writerow(["Product", "Stock"])
            writer.writerows(inventory)

        flash("Inventory updated!", "success")

    return render_template('dashboard.html', inventory=inventory)

# Logout
@app.route('/logout')
def logout():
    session.pop('user', None)
    return redirect(url_for('home'))

# Email Sending Function
def send_email(to_email, subject, message):
    try:
        server = smtplib.SMTP("smtp.gmail.com", 587)
        server.starttls()
        server.login(EMAIL, PASSWORD)
        
        email_message = f"Subject: {subject}\n\n{message}"
        server.sendmail(EMAIL, to_email, email_message)
        server.quit()
        print(f"Email sent to {to_email}")
    except Exception as e:
        print(f"Failed to send email: {e}")

# Inventory Alerts
def check_inventory():
    with open(INVENTORY_FILE, "r") as file:
        reader = csv.reader(file)
        next(reader)  # Skip header
        for row in reader:
            if int(row[1]) < 5:
                send_email(EMAIL, "Low Inventory Alert", f"{row[0]} is running low!")

# Daily Newsletter
def send_newsletter():
    with open(USER_FILE, "r") as file:
        reader = csv.reader(file)
        next(reader)
        for row in reader:
            send_email(row[0], "Daily Newsletter", "Hello! Here’s your daily newsletter.")

# Schedule tasks
def run_scheduler():
    schedule.every().day.at("09:00").do(send_newsletter)
    schedule.every().hour.do(check_inventory)

    while True:
        schedule.run_pending()
        time.sleep(60)

# Start Scheduler in a separate thread
scheduler_thread = threading.Thread(target=run_scheduler)
scheduler_thread.daemon = True
scheduler_thread.start()

# Run Flask app
if __name__ == "__main__":
    app.run(debug=True)
```
