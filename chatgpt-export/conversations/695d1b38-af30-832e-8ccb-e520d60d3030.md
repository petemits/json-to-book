import random

TOPIC_TEMPLATES = {
    "music": {
        "acknowledge": [
            "That makes sense.",

    data = TOPIC_TEMPLATES[topic]
    return (
        random.choice(data["acknowledge"]) + " "
        + random.choice(data["comments"]) + " "

    )

# Endless conversation loop
topic = "music"
print("Conversation started. Type 'exit' to stop.\n")

while True:
    user_input = input("Person A: ")
    if user_input.lower() == "exit":
        break

    reply = generate_reply(topic)
    print("Person B:", reply)

# 🔹 2. FULL PYTHON PROGRAM (PLUG & PLAY)

### ✅ Works in **Pydroid 3**
### ✅ Built-in libraries only

```python
import random
import time

# -----------------------------
# PERSONALITY STYLES
# -----------------------------
PERSONALITIES = {
    "friendly": {
        "ack": ["Yeah,", "Totally,", "That makes sense,", "I get that,"],
        "reflect": ["it’s always interesting.", "that’s relatable.", "people feel that a lot."],
    },
    "curious": {
        "ack": ["Interesting,", "Hmm,", "That’s fascinating,", "I didn’t think of that,"],
        "reflect": ["there’s more behind it.", "it opens more questions.", "it connects to other ideas."],
    },
    "professional": {
        "ack": ["Understood.", "Noted.", "That aligns with common patterns."],
        "reflect": ["it reflects broader trends.", "context matters here.", "there are implications."],
    }
}

# -----------------------------
# TOPIC ENGINE
# -----------------------------
TOPICS = {
    "music": {
        "angles": [
            "live energy",
            "lyrics meaning",
            "crowd interaction",
            "instrument focus",
            "emotional impact"
        ],
        "questions": {
            "live energy": [
                "Did the crowd change how the music felt?",
                "Was the energy intense or relaxed?"
            ],
            "lyrics meaning": [
                "Did any lyrics stick with you?",
                "Do lyrics matter to you live?"
            ],
            "crowd interaction": [
                "Did the artist talk to the audience?",
                "Did it feel personal or distant?"
            ],
            "instrument focus": [
                "Which instrument stood out most?",
                "Do you prefer guitar-heavy music?"
            ],
            "emotional impact": [
                "Did it bring back memories?",
                "Did the music change your mood?"
            ]
        }
    }
}

# -----------------------------
# MEMORY
# -----------------------------
memory = {
    "used_questions": set(),
    "conversation_log": []
}

# -----------------------------
# RESPONSE GENERATOR
# -----------------------------
def generate_response(topic, personality):
    style = PERSONALITIES[personality]
    topic_data = TOPICS[topic]

    angle = random.choice(topic_data["angles"])
    possible_questions = topic_data["questions"][angle]

    question = random.choice(possible_questions)
    while question in memory["used_questions"]:
        question = random.choice(possible_questions)

    memory["used_questions"].add(question)

    response = (
        random.choice(style["ack"]) + " "
        + random.choice(style["reflect"]) + " "
        + question
    )

    memory["conversation_log"].append(response)
    return response

# -----------------------------
# MODE 1: USER CHAT
# -----------------------------
def interactive_chat(topic="music", personality="friendly"):
    print("\nInteractive conversation started (type 'exit' to stop)\n")

    while True:
        user = input("Person A: ")
        if user.lower() == "exit":
            break

        reply = generate_response(topic, personality)
        print("Person B:", reply)

# -----------------------------
# MODE 2: AUTO TWO-PERSON CHAT
# -----------------------------
def auto_dialogue(topic="music", personality="curious", delay=1.5):
    print("\nAuto dialogue started (Ctrl+C to stop)\n")

    while True:
        a = generate_response(topic, personality)
        print("Person A:", a)
        time.sleep(delay)

        b = generate_response(topic, personality)
        print("Person B:", b)
        time.sleep(delay)

# -----------------------------
# RUN
# -----------------------------
if __name__ == "__main__":
    print("Choose mode:")
    print("1 - Interactive chat")
    print("2 - Auto two-person dialogue")

    mode = input("Enter 1 or 2: ")

    if mode == "1":
        interactive_chat(topic="music", personality="friendly")
    else:
        auto_dialogue(topic="music", personality="curious")
```

import csv
import time

# ---------------- PERSONALITY ----------------
PERSONALITY = {
    "friendly": {
        "ack": ["Yeah,", "Totally,", "That makes sense,"],
        "reflect": ["it connects well.", "people relate to that.", "it’s common."],
    },
    "curious": {
        "ack": ["Interesting,", "Hmm,", "That’s intriguing,"],
        "reflect": ["it opens new questions.", "there’s depth there.", "it links ideas."],
    }
}

# ---------------- TOPICS ----------------
TOPICS = {
    "any": {
        "angles": ["experience", "emotion", "impact", "future", "contrast"],
        "questions": [
            "What stood out most?",
            "How did it make you feel?",
            "Did it change your view?",
            "Would you do it again?",
            "How does it compare to before?"
        ]
    }
}

memory = set()
log = []

def generate(personality="friendly"):
    style = PERSONALITY[personality]
    q = random.choice(TOPICS["any"]["questions"])
    while q in memory:
        q = random.choice(TOPICS["any"]["questions"])

    memory.add(q)

    response = (
        random.choice(style["ack"]) + " "
        + random.choice(style["reflect"]) + " "
        + q
    )
    log.append(response)
    return response

# -------- MODE: AUTO TWO AGENTS ----------
def auto_chat():
    while True:
        print("A:", generate("curious"))
        time.sleep(1.5)
        print("B:", generate("friendly"))
        time.sleep(1.5)

# -------- SAVE TO EXCEL ----------
def save_csv():
    with open("conversation.csv", "w", newline="") as f:
        writer = csv.writer(f)
        for line in log:
            writer.writerow([line])

auto_chat()
