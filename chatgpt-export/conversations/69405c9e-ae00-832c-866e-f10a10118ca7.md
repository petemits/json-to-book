# 🧠 FULL GOOGLE APPS SCRIPT (CHATBOT ENGINE)

```javascript
function onUserMessage(userMessage, sessionId) {
  const ss = SpreadsheetApp.getActive();
  const logSheet = ss.getSheetByName("Chat_Log");
  const flowSheet = ss.getSheetByName("Bot_Flows");

  const flows = flowSheet.getDataRange().getValues();
  let botReply = "Sorry, I didn’t understand that. Please try again.";

  for (let i = 1; i < flows.length; i++) {
    if (userMessage.toLowerCase().includes(flows[i][0])) {
      botReply = flows[i][1];
      break;
    }
  }

  logSheet.appendRow([
    new Date(),
    userMessage,
    botReply,
    sessionId
  ]);

  return botReply;
}
```

# 🧩 STEP 4: CREATE GOOGLE CALENDAR APPOINTMENT

```javascript
function createAppointment(sessionId, service, date, time, type, email) {
  const sheet = SpreadsheetApp.getActive().getSheetByName("Appointments");

  const start = new Date(date + " " + time);
  const end = new Date(start.getTime() + 30 * 60000); // 30 min

  const event = CalendarApp.getDefaultCalendar().createEvent(
    service + " Consultation",
    start,
    end,
    { guests: email }
  );

  sheet.appendRow([
    sessionId,
    service,
    date,
    time,
    type,
    event.getId()
  ]);

  return event.getId();
}
```

// Google Sheet Chatbot Script
const SHEET_NAME = "Conversations";
const LOG_SHEET = "Logs";

// Function to get bot response
function getBotResponse(userInput) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getSheetByName(SHEET_NAME);
  const data = sheet.getDataRange().getValues();
  
  let response = "Sorry, I didn't understand that. Can you rephrase?";
  let contactInfo = "";
  let appointmentCTA = "";
  let sessionNote = "";

  // Search for matching trigger
  for (let i = 1; i < data.length; i++) { // start at 1 to skip header
    const trigger = data[i][0];
    if (userInput.toLowerCase().includes(trigger.toLowerCase())) {
      response = data[i][1];
      sessionNote = data[i][2];
      contactInfo = data[i][3];
      appointmentCTA = data[i][4];
      break;
    }
  }

  // Log conversation
  logConversation(userInput, response);

  return {
    response: response,
    session: sessionNote,
    contactInfo: contactInfo,
    appointmentCTA: appointmentCTA
  };
}

// Function to log conversation
function logConversation(userInput, botResponse) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let logSheet = ss.getSheetByName(LOG_SHEET);
  if (!logSheet) {
    logSheet = ss.insertSheet(LOG_SHEET);
    logSheet.appendRow(["Timestamp", "User Input", "Bot Response"]);
  }
  logSheet.appendRow([new Date(), userInput, botResponse]);
}

// Example Web App doPost function for chatbot integration
function doPost(e) {
  const params = JSON.parse(e.postData.contents);
  const userMessage = params.userMessage || "";

  const botData = getBotResponse(userMessage);

  const output = {
    reply: botData.response,
    session: botData.session,
    contactInfo: botData.contactInfo,
    appointmentCTA: botData.appointmentCTA
  };

  return ContentService
    .createTextOutput(JSON.stringify(output))
    .setMimeType(ContentService.MimeType.JSON);
}

const SHEET_NAME = "Conversations";
const LOG_SHEET = "Logs";

/* ---------- UTILITIES ---------- */
function normalize(text) {
  return text.toLowerCase().replace(/[^a-z0-9\s]/g, "").trim();
}

function scoreMatch(input, keywords) {
  let score = 0;
  keywords.forEach(k => {
    if (input.includes(k)) score++;
  });
  return score;
}

/* ---------- INTENT DETECTION ---------- */
function detectIntent(userInput) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME);
  const rows = sheet.getDataRange().getValues();
  const input = normalize(userInput);

  let bestRow = 1;
  let bestScore = 0;

  for (let i = 1; i < rows.length; i++) {
    const keywordCell = rows[i][1]; // Intent Keywords
    if (!keywordCell) continue;

    const keywords = normalize(keywordCell).split(",");
    const score = scoreMatch(input, keywords);

    if (score > bestScore) {
      bestScore = score;
      bestRow = i;
    }
  }
  return bestRow;
}

/* ---------- CHAT RESPONSE ---------- */
function getBotResponse(userMessage) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME);
  const rows = sheet.getDataRange().getValues();

  const row = detectIntent(userMessage);

  const response = rows[row][2];
  const why = rows[row][3];
  const how = rows[row][4];
  const cta = rows[row][5];

  logChat(userMessage, response);

  return {
    reply: response,
    explanation: why,
    help: how,
    cta: cta
  };
}

function logChat(user, bot) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();

  if (!sheet) {

    sheet.appendRow(["Timestamp", "User", "Bot"]);
  }

}

/* ---------- WEB APP ---------- */
function doPost(e) {
  const data = JSON.parse(e.postData.contents);
