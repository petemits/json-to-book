```python
def download_and_extract_repo(zip_url, folder_name):
    zip_path = folder_name + ".zip"

    if not os.path.exists(folder_name):
        print(f"Downloading {folder_name} …")
        urllib.request.urlretrieve(zip_url, zip_path)

        print(f"Extracting {folder_name} …")
        with zipfile.ZipFile(zip_path, 'r') as zip_ref:
            zip_ref.extractall(folder_name)

        print(f"{folder_name} Ready.\n")
    else:
        print(f"{folder_name} already exists.\n")
```

# 4️⃣ Load All TXT / CSV Into Memory

```python
loaded_data = {}

def load_txt(file_path):
    try:
        with open(file_path, "r", encoding="utf-8", errors="ignore") as f:
            return f.read().splitlines()
    except:
        return []

def load_csv(file_path):
    data = []
    try:
        with open(file_path, "r", encoding="utf-8", errors="ignore") as f:
            reader = csv.reader(f)
            headers = next(reader, None)
            for row in reader:
                data.append(row)
        return data
    except:
        return []

# Scan all extracted folders
for folder in datasets.keys():
    folder_path = folder
    loaded_data[folder] = {"txt": [], "csv": []}
    for root, dirs, files in os.walk(folder_path):
        for file in files:
            full_path = os.path.join(root, file)
            if file.endswith(".txt"):
                loaded_data[folder]["txt"].extend(load_txt(full_path))
            elif file.endswith(".csv"):
                loaded_data[folder]["csv"].append(load_csv(full_path))

print("All datasets loaded into memory!")
```

```python
def search_all(keyword):
    results = []
    keyword_lower = keyword.lower()
    for category, types in loaded_data.items():
        # Search TXT
        for line in types["txt"]:
            if keyword_lower in line.lower():
                results.append(f"[{category} TXT] {line}")
        # Search CSV
        for table in types["csv"]:
            for row in table:
                row_text = " ".join(row)
                if keyword_lower in row_text.lower():
                    results.append(f"[{category} CSV] {row_text}")
    return results
```
