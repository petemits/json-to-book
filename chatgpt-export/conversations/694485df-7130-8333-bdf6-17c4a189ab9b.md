## 🔹 OPTION 1: Simple Real-Time Chat (Beginner – No Backend Coding)
Best if you want something **fast** and **working today**.

### Tech
- **Frontend:** HTML + CSS + JavaScript  
- **Backend:** Firebase (real-time database)

### How it works
- Firebase automatically syncs messages in real time
- No server setup required

### Flow
```
User types message
↓
Message saved to Firebase
↓
All connected users instantly receive it
```

### Core Code Example
```html
<input id="msg" placeholder="Type..." />
<button onclick="send()">Send</button>
<div id="chat"></div>

<script src="https://www.gstatic.com/firebasejs/9.23.0/firebase-app.js"></script>
<script src="https://www.gstatic.com/firebasejs/9.23.0/firebase-database.js"></script>

<script>
const app = firebase.initializeApp({
  apiKey: "YOUR_KEY",
  databaseURL: "YOUR_DB_URL"
});

const db = firebase.database();

function send() {
  db.ref("chat").push({
    text: msg.value,
    time: Date.now()
  });
}

db.ref("chat").on("child_added", snap => {
  chat.innerHTML += `<p>${snap.val().text}</p>`;
});
</script>
```

✅ **Real-time**
✅ **Scales automatically**
❌ Requires Firebase account

function getTodayWeather() {

  // ====== CHANGE LAT / LON IF NEEDED ======
  // Toronto example
  var latitude = 43.7;
  var longitude = -79.4;

  var url = "https://api.open-meteo.com/v1/forecast"
          + "?latitude=" + latitude
          + "&longitude=" + longitude
          + "&current_weather=true";

  var response = UrlFetchApp.fetch(url);
  var data = JSON.parse(response.getContentText());

  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();

  var temp = data.current_weather.temperature;
  var wind = data.current_weather.windspeed;
  var code = data.current_weather.weathercode;
  var time = new Date();

  sheet.getRange("B2").setValue(temp);
  sheet.getRange("C2").setValue(wind);
  sheet.getRange("D2").setValue(code);
  sheet.getRange("E2").setValue(getCondition(code));
  sheet.getRange("F2").setValue(time);
}

// ===== WEATHER CODE TO TEXT (CNN STYLE) =====
function getCondition(code) {

  if (code === 0) return "Clear";
  if (code <= 3) return "Partly Cloudy";
  if (code <= 48) return "Fog";
  if (code <= 67) return "Rain";
  if (code <= 77) return "Snow";
  if (code <= 82) return "Heavy Rain";
  if (code <= 99) return "Storm";

  return "Unknown";
}

// ===== AUTO REFRESH (OPTIONAL) =====
function autoUpdateHourly() {
  ScriptApp.newTrigger("getTodayWeather")
    .timeBased()
    .everyHours(1)
    .create();
}
```

/*************** PLUG & PLAY WEATHER SYSTEM ****************/

function onOpen() {
  SpreadsheetApp.getUi()
    .createMenu("🌤 Weather")
    .addItem("Update Weather Now", "updateWeather")
    .addItem("Auto Update Hourly", "enableAutoUpdate")
    .addToUi();
}

function updateWeather() {

  const sheet = SpreadsheetApp.getActiveSheet();
  const city = sheet.getRange("A2").getValue();

  if (!city) {
    SpreadsheetApp.getUi().alert("Please enter a city in cell A2");
    return;
  }

  // ---- GEOCODING (City → Lat/Lon) ----
  const geoUrl = "https://geocoding-api.open-meteo.com/v1/search?name="
                 + encodeURIComponent(city) + "&count=1";

  const geoRes = UrlFetchApp.fetch(geoUrl);
  const geoData = JSON.parse(geoRes.getContentText());

  if (!geoData.results || geoData.results.length === 0) {
    SpreadsheetApp.getUi().alert("City not found");
    return;
  }

  const lat = geoData.results[0].latitude;
  const lon = geoData.results[0].longitude;

  // ---- WEATHER DATA ----
  const weatherUrl = "https://api.open-meteo.com/v1/forecast"
                   + "?latitude=" + lat
                   + "&longitude=" + lon
                   + "&current_weather=true";

  const weatherRes = UrlFetchApp.fetch(weatherUrl);
  const weather = JSON.parse(weatherRes.getContentText()).current_weather;

  sheet.getRange("B2").setValue(weather.temperature);
  sheet.getRange("C2").setValue(weather.windspeed);
  sheet.getRange("D2").setValue(mapCondition(weather.weathercode));
  sheet.getRange("E2").setValue(new Date());
}

function mapCondition(code) {
  if (code === 0) return "Clear ☀️";
  if (code <= 3) return "Partly Cloudy ⛅";
  if (code <= 48) return "Fog 🌫️";
  if (code <= 67) return "Rain 🌧️";
  if (code <= 77) return "Snow ❄️";
  if (code <= 82) return "Heavy Rain ⛈️";
  return "Storm 🌩️";
}

function enableAutoUpdate() {
  ScriptApp.getProjectTriggers().forEach(t => ScriptApp.deleteTrigger(t));

  ScriptApp.newTrigger("updateWeather")
    .timeBased()
    .everyHours(1)
    .create();

  SpreadsheetApp.getUi().alert("Hourly auto-update enabled");
}
```

function setupWeatherSystem() {

  // CREATE SPREADSHEET
  const sheetFile = SpreadsheetApp.create("🌤 Live Weather Dashboard");
  const sheet = sheetFile.getActiveSheet();

  // SET HEADERS
  sheet.getRange("A1:F1").setValues([[
    "City",
    "Temperature (°C)",
    "Wind (km/h)",
    "Condition",
    "Last Updated",
    "Status"
  ]]);

  // DEFAULT CITY
  sheet.getRange("A2").setValue("Toronto");

  // FORMAT
  sheet.getRange("A1:F1").setFontWeight("bold");
  sheet.setFrozenRows(1);
  sheet.autoResizeColumns(1, 6);

  // ADD MENU
  SpreadsheetApp.setActiveSpreadsheet(sheetFile);
  onOpen();

  // FIRST WEATHER LOAD
  updateWeather();

  // OPEN SHEET
  SpreadsheetApp.getUi().alert(
    "Weather system created!\n\nUse menu: 🌤 Weather → Update Weather"
  );

  return sheetFile.getUrl();
}

function onOpen() {
  SpreadsheetApp.getUi()
    .createMenu("🌤 Weather")
    .addItem("Update Weather Now", "updateWeather")
    .addItem("Enable Hourly Auto-Update", "enableAutoUpdate")
    .addToUi();

function updateWeather() {

  const sheet = SpreadsheetApp.getActiveSheet();
  const city = sheet.getRange("A2").getValue();

  if (!city) {
    sheet.getRange("F2").setValue("City missing");
    return;
  }

  try {
    // CITY → LAT / LON
    const geoUrl = "https://geocoding-api.open-meteo.com/v1/search?name="
      + encodeURIComponent(city) + "&count=1";

    const geoData = JSON.parse(UrlFetchApp.fetch(geoUrl).getContentText());

    if (!geoData.results) {
      sheet.getRange("F2").setValue("City not found");
      return;
    }

    const lat = geoData.results[0].latitude;
    const lon = geoData.results[0].longitude;

    // WEATHER DATA
    const weatherUrl = "https://api.open-meteo.com/v1/forecast"
      + "?latitude=" + lat
      + "&longitude=" + lon
      + "&current_weather=true";

    const weather = JSON.parse(
      UrlFetchApp.fetch(weatherUrl).getContentText()
    ).current_weather;

    sheet.getRange("B2").setValue(weather.temperature);
    sheet.getRange("C2").setValue(weather.windspeed);
    sheet.getRange("D2").setValue(mapCondition(weather.weathercode));
    sheet.getRange("E2").setValue(new Date());
    sheet.getRange("F2").setValue("Live");

  } catch (e) {
    sheet.getRange("F2").setValue("Error");
  }
}

function mapCondition(code) {
  if (code === 0) return "Clear ☀️";
  if (code <= 3) return "Partly Cloudy ⛅";
  if (code <= 48) return "Fog 🌫️";
  if (code <= 67) return "Rain 🌧️";
  if (code <= 77) return "Snow ❄️";
  if (code <= 82) return "Heavy Rain ⛈️";

    .timeBased()
