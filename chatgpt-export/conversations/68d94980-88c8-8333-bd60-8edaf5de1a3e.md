# 1) High-level architecture (text diagram)

PSTN ⟶ 3CX PBX (route/DID)  
→ internal transfer / forward ⟶ AI extension (SIP) — **this is the Python SIP agent** (PJSIP)  
AI SIP agent:
- receives RTP audio from 3CX
- captures short audio chunks → Whisper STT (local)
- sends text + context → Ollama local LLM (local REST)
- receives LLM reply → Bark TTS (local) → plays TTS audio back into the call via PJSIP
- logs transcripts, events, and structured outputs (JSON) and exposes them via Flask APIs/UI

For outbound calls the AI agent can `INVITE` via 3CX (dial through PBX/trunk) or call 3CX Call Control API to create calls programmatically then bridge to the AI extension.

Why this approach? Registering the agent as a SIP extension keeps 3CX configuration simple (route calls or transfer to the agent extension). PJSIP gives media control (record/play WAV). (Docs: 3CX Call Control; PJSIP media). citeturn0search1turn2search1

### `src/stt.py` — Whisper wrapper (simple)
```python
# src/stt.py
import whisper
import os

MODEL_CACHE = {}

def load_whisper(model_name="small"):
    if model_name in MODEL_CACHE:
        return MODEL_CACHE[model_name]
    model = whisper.load_model(model_name)
    MODEL_CACHE[model_name] = model
    return model

def transcribe_wav(path, model_name="small", language="en"):
    model = load_whisper(model_name)
    # whisper returns dict with "text"
    result = model.transcribe(path, language=language)
    return result.get("text", "").strip()
```
(If you need faster CPU inference use `faster-whisper` or a small/quantized model; see faster-whisper docs.) citeturn1search1

### `src/llm.py` — Ollama client (sync)
```python
# src/llm.py
import requests
import json
from typing import List

OLLAMA_URL = "http://localhost:11434"  # default when running ollama serve

def llm_chat(messages: List[dict], model="llama2"):
    """
    messages = [{"role":"system","content":"..."}, {"role":"user","content":"..."}]
    """
    url = f"{OLLAMA_URL}/api/chat"
    payload = {"model": model, "messages": messages, "stream": False}
    resp = requests.post(url, json=payload, timeout=60)
    resp.raise_for_status()
    data = resp.json()
    # Ollama returns an array of choices/outputs depending on endpoint; adapt if streaming
    return data.get("content") or data  # adapt as needed

def generate(prompt: str, model="llama2"):
    url = f"{OLLAMA_URL}/api/generate"
    payload = {"model": model, "prompt": prompt, "stream": False}
    r = requests.post(url, json=payload, timeout=60)
    r.raise_for_status()
    return r.json().get("text") or r.json()
```
(Ollama exposes `/api/chat` and `/api/generate` on localhost by default.) citeturn0search2turn0search10

### `src/tts.py` — Bark via Transformers pipeline (simple)
```python
# src/tts.py
import scipy.io.wavfile as wavfile
from transformers import pipeline

_TTS_PIPE = None

def load_bark(model="suno/bark"):
    global _TTS_PIPE
    if _TTS_PIPE is None:
        _TTS_PIPE = pipeline("text-to-speech", model)
    return _TTS_PIPE

def tts_to_wav(text, out_path="out_tts.wav"):
    pipe = load_bark()
    res = pipe(text, forward_params={"do_sample": True})
    rate = res["sampling_rate"]
    data = res["audio"]
    wavfile.write(out_path, rate, data)
    return out_path
```
(You can also use the Bark repo’s native code for more control or voice-consistency.) citeturn4search1

### `src/sip_agent.py` — main SIP loop (PJSUA)
```python
# src/sip_agent.py
import pjsua as pj
import time
import threading
import os
import uuid
from stt import transcribe_wav
from llm import llm_chat
from tts import tts_to_wav
from datetime import datetime

CHUNK_SECONDS = 4
CALL_RECORD_DIR = "tmp_calls"
os.makedirs(CALL_RECORD_DIR, exist_ok=True)

class MyCallCallback(pj.CallCallback):
    def __init__(self, call, app):
        pj.CallCallback.__init__(self, call)
        self.app = app
        self.player_id = None

    def on_state(self):
        print("Call state:", self.call.info().state_text)
        if self.call.info().state == pj.CallState.DISCONNECTED:
            print("Call disconnected.")

    def on_media_state(self):
        if self.call.info().media_state == pj.MediaState.ACTIVE:
            # spawn worker thread to handle conversation loop for this call
            threading.Thread(target=self.handle_call_media, daemon=True).start()

    def handle_call_media(self):
        call = self.call
        # connect call audio to local handling via per-chunk recorders
        while call.info().state == pj.CallState.CONFIRMED or call.info().state == pj.CallState.EARLY:
            fname = os.path.join(CALL_RECORD_DIR, f"{datetime.utcnow().strftime('%Y%m%d%H%M%S')}_{uuid.uuid4().hex}.wav")
            rec_id = pj.Lib.instance().create_recorder(fname)
            rec_slot = pj.Lib.instance().recorder_get_slot(rec_id)
            call_slot = call.info().conf_slot
            # connect call -> recorder to capture remote audio
            pj.Lib.instance().conf_connect(call_slot, rec_slot)
            time.sleep(CHUNK_SECONDS)
            # stop recording for this chunk
            pj.Lib.instance().conf_disconnect(call_slot, rec_slot)
            pj.Lib.instance().destroy_recorder(rec_id)
            # do STT
            try:
                text = transcribe_wav(fname)
                print("STT:", text)
                if text.strip():
                    # Build messages for LLM - a minimal example:
                    messages = [
                        {"role":"system","content":"You are a helpful sales/booking assistant for a steel manufacturer. Ask for what you need: company name, part, quantity, delivery date, phone number."},
                        {"role":"user","content": text}
                    ]
                    reply = llm_chat(messages)
                    if isinstance(reply, dict):
                        # adapt depending on Ollama response format
                        reply_text = reply.get("content") or reply.get("text") or str(reply)
                    else:
                        reply_text = str(reply)
                    print("LLM reply:", reply_text)
                    # TTS
                    tts_path = tts_to_wav(reply_text)
                    # play TTS into the call
                    player_id = pj.Lib.instance().create_player(tts_path, loop=False)
                    player_slot = pj.Lib.instance().player_get_slot(player_id)
                    pj.Lib.instance().conf_connect(player_slot, call_slot)
                    # wait for TTS to finish (simple sleep = length of file). For production use events.
                    time.sleep( max(1, (CHUNK_SECONDS)) )
                    pj.Lib.instance().conf_disconnect(player_slot, call_slot)
                    pj.Lib.instance().player_destroy(player_id)
            except Exception as e:
                print("Error stt/llm/tts:", e)
        print("Call handling loop ended.")
```

**How to start the SIP agent (example):**
```python
# src/run_agent.py
import pjsua as pj
from sip_agent import MyCallCallback
import time, sys

SIP_DOMAIN = "pbx.example.local"
SIP_USER = "9100"
SIP_PASS = "password"
SIP_PROXY = "sip:pbx.example.local"

lib = pj.Lib()
try:
    lib.init(log_cfg = pj.LogConfig(level=3, callback=None))
    transport = lib.create_transport(pj.TransportType.UDP, pj.TransportConfig(5060))
    lib.start()
    acc_cfg = pj.AccountConfig(domain=SIP_DOMAIN, username=SIP_USER, password=SIP_PASS)
    acc_cfg.id = f"sip:{SIP_USER}@{SIP_DOMAIN}"
    acc_cfg.reg_uri = f"sip:{SIP_DOMAIN}"
    account = lib.create_account(acc_cfg)

    print("SIP agent registered. Waiting for calls...")
    # keep process running
    while True:
        time.sleep(1)
except KeyboardInterrupt:
    print("Bye")
finally:
    lib.destroy()
    lib = None
```

**Notes & caveats:**
- Exact API names vary with pjsua vs pjsua2; use the version you built. See the PJSIP docs for `create_recorder`, `player_get_slot`, `conf_connect`, etc. citeturn2search1
- This sample records short chunks and processes them one by one. For low latency you’ll want streaming transcription and an RTP-level media pipeline rather than file chunks.
- Always test with a stub (playback file) before connecting to live PSTN calls.

### `src/webapp.py` — Flask control & logs (simple)
```python
# src/webapp.py
from flask import Flask, request, jsonify, send_file
import json, os

app = Flask(__name__)
LOG_FILE = "logs/calls.log"

def append_log(obj):
    os.makedirs(os.path.dirname(LOG_FILE), exist_ok=True)
    with open(LOG_FILE,"a",encoding="utf8") as f:
        f.write(json.dumps(obj, default=str) + "\n")

@app.route("/api/logs/recent")
def recent_logs():
    if not os.path.exists(LOG_FILE):
        return jsonify([])
    with open(LOG_FILE,"r",encoding="utf8") as f:
        lines = f.readlines()[-200:]
    return jsonify([json.loads(l) for l in lines])

@app.route("/api/make_call", methods=["POST"])
def make_call():
    data = request.json
    target = data.get("target")
    # for security, validate
    if not target:
        return jsonify({"error":"target required"}), 400
    # In production call into SIP agent via inter-process API to make outgoing call
    append_log({"event":"outbound_call_requested","target":target})
    # placeholder: respond OK; actual call uses sip_agent.Account.make_call(...)
    return jsonify({"status":"queued","target":target})

if __name__ == "__main__":
    app.run(host="0.0.0.0", port=5000)
```

You can wire this Flask app to call `account.make_call()` in the running pjsua process via an IPC method (HTTP between processes, local Unix socket, or `multiprocessing` shared object). I recommend running the SIP agent and Flask server in the same process or use a lightweight IPC (Flask→SIP agent REST).

# 6) Deployment steps — step-by-step (Linux first, Windows alternatives after)

## A — Linux / WSL2 (recommended)
1. Prepare machine (Ubuntu 22.04+):
   - `sudo apt update && sudo apt install -y build-essential libasound2-dev swig ffmpeg python3 python3-venv python3-pip`
   - Install `curl`, `git`.
2. Install Ollama (follow Ollama install instructions) and `ollama serve`. Confirm `http://localhost:11434/api` works. citeturn0search10
3. Create Python venv:
   ```bash
   python3 -m venv venv
   . venv/bin/activate
   pip install --upgrade pip
   pip install openai-whisper transformers scipy requests flask
   # or faster-whisper for speed:
   pip install faster-whisper
   ```
4. Install Bark (via Hugging Face or GitHub instructions). Example:
   ```bash
   pip install transformers scipy
   ```
   and test `from transformers import pipeline; pipeline("text-to-speech","suno/bark")`.
   (Download model weights as instructed by HuggingFace or `git clone` the repo.) citeturn4search1
5. Install PJSIP + Python wrapper:
   - Follow PJSIP build docs: configure, `make dep`, `./configure`, `make clean && make`, build `pjsua` and the Python SWIG module (pjsua2/pjsua). On Linux this is documented and much easier than Windows. See PJSIP build docs. citeturn0search8turn2search1
6. Copy the `steel-ai-voice-agent/` tree to the server, edit `config.yaml` with PBX SIP host, extension, and model paths.
7. Start required servers:
   - `ollama serve` (Ollama)
   - (If using faster-whisper server wrappers you installed) start that
   - Start Flask `python src/webapp.py`
   - Start SIP agent: `python src/run_agent.py`
8. Test with internal 3CX extension: call the DID/route point and transfer to the AI extension. Watch logs and `logs/calls.log`.

## B — Windows (EXE) — notes & steps
**Important:** building the pjsua/pjsua2 Python module on Windows is nontrivial and has community issues; many users build pjsua on Linux then run via WSL2. See community issue threads. citeturn5search5turn0search12

If you insist on Windows native:
1. Install Visual Studio Build Tools (MSVC), SWIG for Windows, Python 3.10 (64-bit).  
2. Build PJSIP from source with MSVC and create Python SWIG bindings. Follow PJSIP Windows build docs; expect trouble and long compile times. citeturn0search8
3. Install Python deps: `pip install openai-whisper transformers scipy requests flask pyinstaller`
4. Run the same source code; audio device handling on Windows via PJSIP will work once the wrapper is successfully built and installed.
5. Build EXE with PyInstaller:
   ```bat
   pyinstaller --name steel_ai_agent --onefile --add-data "config.yaml;." --hidden-import=pjsua src/run_agent.py
   ```
   You will likely need to tweak a `.spec` file to include pjsip DLLs, model file directories, and other native libs. See `scripts/build_exe.bat` for a sample.  
**Alternative (strongly recommended):** run heavy ML + SIP on Linux and create a small Windows EXE that talks to the Linux agent's HTTP API (Flask). This avoids packaging models and PJSIP on Windows and keeps heavy compute on a server.

# 10) Security, privacy & compliance (must read)

```python
agent_index = 0
agent_lock = threading.Lock()
agent_health = {a["id"]: {"healthy": True, "last_check": None} for a in AGENTS}

def choose_agent():
    global agent_index
    with agent_lock:
        n = len(AGENTS)
        if n == 0:
            return None
        # simple round-robin but skip unhealthy agents
        for i in range(n):
            idx = (agent_index + i) % n
            a = AGENTS[idx]
            status = agent_health.get(a["id"], {})
            if status.get("healthy", True):
                agent_index = (idx + 1) % n
                return a
        # if all unhealthy, fallback to the first agent
        agent_index = (agent_index + 1) % n
        return AGENTS[agent_index]

    Your PBX must be configured to POST the call-start (or ringing) event to this URL.
    Expected JSON (example):

def health_check_loop():
    while True:
        for a in AGENTS:
            hid = a["id"]
            url = a.get("health_url")
            ok = False
            if url:
                try:
                    r = requests.get(url, timeout=3)
                    ok = (r.status_code == 200)
                except Exception:
                    ok = False
            else:
                # if no health_url assume healthy
                ok = True
            agent_health[hid] = {"healthy": ok, "last_check": time.time()}
            logging.debug("Health %s => %s", hid, ok)
```

If you prefer performing transfers from a SIP client (e.g., a controller that holds the call), you can use PJSIP to send a REFER to transfer the call. This file is a *starting template* — it is not fully runnable without a pjsua build and mapping between PBX call ID and pjsua.Call instance.

```python
# transfer_sip.py
# NOTE: this is illustrative only. You must build pjsua & run a SIP account that is in the call path.
import pjsua as pj
import logging

lib = None

def init_pjsua(local_port=5065):
    global lib
    lib = pj.Lib()
    lib.init()
    tcfg = pj.TransportConfig(local_port)
    lib.create_transport(pj.TransportType.UDP, tcfg)
    lib.start()
    logging.info("PJSUA started")

def sip_refer(call_obj, target_sip):
    """
    call_obj: pjsua.Call instance representing the active call
    target_sip: string like "sip:9101@pbx.example.local"
    """
    if not call_obj:
        raise RuntimeError("call_obj required")
    # Send REFER to transfer the call
    try:
        call_obj.xfer(target_sip)  # pjsua high-level API
        logging.info("Issued REFER to %s", target_sip)
    except Exception as e:
        logging.exception("REFER failed: %s", e)
        raise
```

**Notes:**
- `call_obj.xfer(target)` sends SIP REFER (or re-INVITE depending on library). You must already have the pjsua `Call` object for the call you want to transfer. Mapping PBX call IDs to pjsua call objects usually requires the controller to be the media endpoint that received the call or to be in the call path.
- Many PBX systems expose REST Call Control APIs which are easier — they identify calls by call_id and let you request a transfer.

```python
import whisper
import sounddevice as sd
import numpy as np
import ollama
import torch
from bark import generate_audio, preload_models

# Load models
model = whisper.load_model("base")
preload_models()

def process_audio(audio_chunk):
    # Save temporary WAV
    audio_array = np.frombuffer(audio_chunk, dtype=np.int16).astype(np.float32) / 32768.0
    text = model.transcribe(audio_array, fp16=False)["text"]
    print("Heard:", text)

    # Generate response
    response = ollama.chat("llama3", [{"role":"user","content":text}])
    reply = response["message"]["content"]
    print("AI says:", reply)

    # Generate voice
    audio_out = generate_audio(reply, history_prompt="en_speaker_1")
    sd.play(audio_out, samplerate=24000)
    sd.wait()
```

```python
# src/stt.py
import os
import numpy as np
from faster_whisper import WhisperModel  # falls back to whisper if not installed
from pathlib import Path

MODEL_CACHE = {}

def load_model(name="small"):
    if name in MODEL_CACHE:
        return MODEL_CACHE[name]
    # faster-whisper autodetects device
    m = WhisperModel(name, device="cpu", compute_type="int8_float16")
    MODEL_CACHE[name] = m
    return m

def transcribe_wav(wav_path, model_name="small", language="en"):
    """
    Return text for a given WAV file path using faster-whisper.
    """
    model = load_model(model_name)
    segments, info = model.transcribe(wav_path, beam_size=5, language=language)
    text = " ".join([s.text for s in segments])
    return text.strip()
```

```python
def generate_chat_reply(user_text: str, model="llama3", system_prompt=None, timeout=60):
    """

    # Basic prompt – wrap with system prompt if provided
    prompt = ""

    prompt += user_text
    payload = {"model": model, "prompt": prompt, "max_tokens": 512, "temperature": 0.2}
    resp = requests.post(url, json=payload, timeout=timeout)
    resp.raise_for_status()
    data = resp.json()
    # Ollama returns 'text' or content depending on version
    if isinstance(data, dict) and "text" in data:
        return data["text"]
    # fallback: attempt to extract from structure
    return str(data)
```

# 7) `src/sip_agent.py` — main SIP agent (PJSUA)
This module contains the heavy lifting: register to 3CX, accept calls, capture short WAV chunks from the call, do STT/LLM/TTS, and play audio back.

```python
# src/sip_agent.py
import pjsua as pj
import time
import threading
import os
import uuid
from datetime import datetime
import wave
import audioop

from stt import transcribe_wav
from llm import generate_chat_reply
from tts import text_to_wav_bark
from utils import append_log
import yaml

# Load config
CONFIG_PATH = os.path.join(os.path.dirname(__file__), "..", "config.yaml")
with open(CONFIG_PATH, "r") as f:
    cfg = yaml.safe_load(f)

CHUNK_SECONDS = cfg.get("audio", {}).get("chunk_seconds", 4)
SAMPLE_RATE = cfg.get("audio", {}).get("sample_rate", 16000)
TMP_DIR = cfg.get("audio", {}).get("tmp_dir", "tmp_calls")
os.makedirs(TMP_DIR, exist_ok=True)

class AgentCallCallback(pj.CallCallback):
    def __init__(self, call, app):
        pj.CallCallback.__init__(self, call)
        self.app = app
        self.recorder_id = None
        self.running = True

    def on_state(self):
        print("Call state:", self.call.info().state_text)
        if self.call.info().state == pj.CallState.DISCONNECTED:
            self.running = False
            print("Call disconnected")

    def on_media_state(self):
        if self.call.info().media_state == pj.MediaState.ACTIVE:
            print("Media is active - starting capture thread")
            t = threading.Thread(target=self.media_loop, daemon=True)
            t.start()

    def media_loop(self):
        """
        Record small WAV chunks from the call, process them, and play TTS reply.
        """
        call = self.call
        lib = pj.Lib.instance()
        call_slot = call.info().conf_slot

        while call.info().state == pj.CallState.CONFIRMED and self.running:
            fname = os.path.join(TMP_DIR, f"{datetime.utcnow().strftime('%Y%m%d%H%M%S')}_{uuid.uuid4().hex}.wav")
            try:
                # create recorder to file with appropriate sample rate
                rec_id = lib.create_recorder(fname, pcm_format=pj.PJMEDIA_FORMAT_PCM16)
                rec_slot = lib.recorder_get_slot(rec_id)
                # connect call -> recorder (remote audio into file)
                lib.conf_connect(call_slot, rec_slot)
                time.sleep(CHUNK_SECONDS)
                # disconnect and destroy recorder
                lib.conf_disconnect(call_slot, rec_slot)
                lib.recorder_destroy(rec_id)
            except Exception as e:
                print("Recorder error:", e)
                break

            # ensure file exists and size > small threshold
            try:
                text = transcribe_wav(fname, model_name=cfg["models"]["whisper_model"])
                print("Transcribed:", text)
                if text.strip():
                    # LLM reply
                    system_prompt = ("You are a helpful sales/booking assistant for a steel manufacturer. "
                                     "Ask for company name, contact, phone, email, product grade, quantity, dimensions, delivery date.")
                    reply = generate_chat_reply(text, model=cfg["models"]["llm_model"], system_prompt=system_prompt)
                    print("LLM reply:", reply)
                    # TTS -> WAV file
                    tts_out = fname.replace(".wav", "_tts.wav")
                    # Use Bark - implement text_to_wav_bark
                    text_to_wav_bark(reply, tts_out, voice=cfg["models"].get("tts_voice"))
                    # play TTS into call via pjsua player
                    player_id = lib.create_player(tts_out, loop=False)
                    player_slot = lib.player_get_slot(player_id)
                    lib.conf_connect(player_slot, call_slot)
                    # approximate wait: get file duration or just wait chunk_seconds
                    time.sleep(CHUNK_SECONDS + 0.5)
                    lib.conf_disconnect(player_slot, call_slot)
                    lib.player_destroy(player_id)
                    # Log call info (append transcript/response)
                    append_log({"call_id": call.info().id, "transcript": text, "reply": reply})
            except Exception as e:
                print("Processing error:", e)

        print("Exiting media loop for call")
```

Notes on `create_recorder` options: API differs slightly across pjsua versions. The code uses `create_recorder(fname)` and `recorder_get_slot(rec_id)` then `conf_connect`. Adjust if your pjsua variant uses different function signatures.

```python
from av import AudioFrame

class OutgoingAudioTrack(MediaStreamTrack):
    kind = "audio"
    def __init__(self, sample_rate=CFG["sample_rate"]):
        super().__init__()  # don't forget this!
        self.sample_rate = sample_rate
        self.queue = asyncio.Queue()
        self.silence_frame = None

    async def recv(self):
        """
        Wait for PCM frames queued by the TTS thread. If none available, return silence frames.
        We'll produce 20ms frames of mono 16-bit PCM at sample_rate
        """
        # Attempt to get next chunk (block up to 1s)
        try:
            pcm, sr = await asyncio.wait_for(self.queue.get(), timeout=1.0)
        except asyncio.TimeoutError:
            # produce small silence frame
            pcm = np.zeros(int(self.sample_rate * 0.02), dtype=np.int16)
            sr = self.sample_rate

        # convert to AudioFrame (aiortc expects float32 PCM, but av.AudioFrame can handle)
        frame = AudioFrame.from_ndarray(pcm.astype(np.int16), layout="mono")
        frame.sample_rate = sr
        frame.pts = None
        frame.time_base = None
        await asyncio.sleep(0)  # yield
        return frame

    async def push_pcm(self, pcm: np.ndarray, sr: int):
        await self.queue.put((pcm, sr))

class SIPWebSocketClient:
    def __init__(self, cfg):
        self.cfg = cfg
        self.ws = None
        self.callid_to_pc: Dict[str, RTCPeerConnection] = {}
        self.dialogs = {}
        self.cseq = 1
        self.branch = lambda: "z9hG4bK-" + uuid.uuid4().hex
        self.tag = lambda: uuid.uuid4().hex[:8]

        self.transport = "WS"
        self.authorization_header = None
        self.wss_url = cfg["wss_url"]
        self.insecure = cfg.get("insecure", True)
        # SIP identity
        self.user = cfg["ext"]
        self.domain = cfg["sip_domain"]
        self.password = cfg["password"]
        self.call_count = 0

    async def connect(self):
        sslctx = None
        if self.wss_url.startswith("wss://"):
            sslctx = ssl.create_default_context()
            if self.insecure:
                sslctx.check_hostname = False
                sslctx.verify_mode = ssl.CERT_NONE
        print("Connecting to WSS:", self.wss_url, "insecure:", self.insecure)
        self.ws = await websockets.connect(self.wss_url, ssl=sslctx, extra_headers={"User-Agent": self.cfg["user_agent"]})
        # Do initial REGISTER
        await self.register()

        # listen loop
        asyncio.ensure_future(self._receive_loop())

    async def _receive_loop(self):
        async for raw in self.ws:
            # raw is a text SIP message (3CX sends SIP messages over ws)
            # print("<<", raw[:200])
            start, headers, body = parse_sip_message(raw)
            # handle responses and requests (INVITE, 200 OK, BYE, etc)
            # If start line begins with "SIP/2.0" it's a response
            if start.startswith("SIP/2.0"):
                code = int(start.split()[1])
                # handle 401 for REGISTER, 200 OK for INVITE, etc.
                if code == 401:
                    # get WWW-Authenticate header
                    www = headers.get("WWW-Authenticate") or headers.get("Proxy-Authenticate")
                    await self._handle_401(www, raw)
                elif code == 200:
                    # Could be REGISTER or INVITE response; we ignore here
                    pass
                else:
                    pass
            else:
                # Request
                method = start.split()[0]
                if method == "INVITE":
                    asyncio.ensure_future(self._handle_invite(headers, body, raw))
                elif method == "ACK":
                    # ACK after our 200 OK
                    pass
                elif method == "BYE":
                    # tear down call
                    call_id = headers.get("Call-ID")
                    print("BYE received for", call_id)
                    await self._hangup(call_id)
                else:
                    print("Unhandled SIP request:", method)

    async def _hangup(self, call_id):
        pc = self.callid_to_pc.get(call_id)
        if pc:
            try:
                await pc.close()
            except:
                pass
            del self.callid_to_pc[call_id]

    async def register(self):
        # send REGISTER without auth, expect 401
        req = []
        via = f"SIP/2.0/WS {self.domain};branch={self.branch()}"
        req.append(f"REGISTER sip:{self.domain} SIP/2.0")
        req.append(f"Via: {via}")
        req.append(f"From: <sip:{self.user}@{self.domain}>;tag={self.tag()}")
        req.append(f"To: <sip:{self.user}@{self.domain}>")
        req.append(f"Contact: <sip:{self.user}@{self.domain}>")
        req.append(f"Call-ID: {uuid.uuid4().hex}")
        req.append(f"CSeq: {self.cseq} REGISTER")
        req.append(f"Max-Forwards: 70")
        req.append(f"User-Agent: {self.cfg['user_agent']}")
        req.append(f"Content-Length: 0")
        msg = "\r\n".join(req) + "\r\n\r\n"
        await self.ws.send(msg)
        print("REGISTER sent (no auth)")

    async def _handle_401(self, www_header: Optional[str], raw: str):
        # parse WWW-Authenticate: Digest realm="...", nonce="..."
        if not www_header:
            print("401 received but no WWW-Authenticate header")
            return
        realm_m = re.search(r'realm="([^"]+)"', www_header)
        nonce_m = re.search(r'nonce="([^"]+)"', www_header)
        realm = realm_m.group(1) if realm_m else ""
        nonce = nonce_m.group(1) if nonce_m else ""
        uri = f"sip:{self.domain}"
        response, cnonce = compute_digest_response(self.user, self.password, realm, "REGISTER", uri, nonce)
        auth = ('Digest username="%s", realm="%s", nonce="%s", uri="%s", response="%s", '
                'algorithm=MD5, qop=auth, nc=00000001, cnonce="%s"' %
                (self.user, realm, nonce, uri, response, cnonce))
        # now send REGISTER with Authorization
        self.cseq += 1
        req = []
        via = f"SIP/2.0/WS {self.domain};branch={self.branch()}"
        req.append(f"REGISTER sip:{self.domain} SIP/2.0")
        req.append(f"Via: {via}")
        req.append(f"From: <sip:{self.user}@{self.domain}>;tag={self.tag()}")
        req.append(f"To: <sip:{self.user}@{self.domain}>")
        req.append(f"Contact: <sip:{self.user}@{self.domain}>")
        req.append(f"Call-ID: {uuid.uuid4().hex}")
        req.append(f"CSeq: {self.cseq} REGISTER")
        req.append(f"Max-Forwards: 70")
        req.append(f"User-Agent: {self.cfg['user_agent']}")
        req.append(f"Authorization: {auth}")
        req.append(f"Content-Length: 0")
        msg = "\r\n".join(req) + "\r\n\r\n"
        await self.ws.send(msg)
        print("REGISTER sent with Authorization")

    async def _handle_invite(self, headers, body, raw):
        # When 3CX sends an INVITE, it includes SDP in body
        call_id = headers.get("Call-ID")
        from_hdr = headers.get("From")
        via = headers.get("Via")
        cseq = headers.get("CSeq")
        print("INVITE received. Call-ID:", call_id)
        # create RTCPeerConnection
        ice_servers = [RTCIceServer(urls=CFG["stun_servers"])] if CFG.get("stun_servers") else None
        config = RTCConfiguration(iceServers=[{"urls": s} for s in CFG["stun_servers"]]) if CFG.get("stun_servers") else None
        pc = RTCPeerConnection(configuration=config)
        outgoing = OutgoingAudioTrack(sample_rate=CFG["sample_rate"])
        pc.addTrack(outgoing)

        @pc.on("track")
        def on_track(track):
            print("Incoming track kind:", track.kind)
            if track.kind == "audio":
                # process incoming audio: write to temporary file in chunks
                asyncio.ensure_future(self._consume_remote_audio(track, call_id))

        # set remote description with the SDP offer (body)
        await pc.setRemoteDescription(RTCSessionDescription(sdp=body, type="offer"))
        # create answer
        answer = await pc.createAnswer()
        await pc.setLocalDescription(answer)

        # send 200 OK with SDP answer in body
        self.cseq += 1
        sr = []
        sr.append(f"SIP/2.0 200 OK")
        sr.append(f"Via: {via}")
        sr.append(f"From: {from_hdr}")
        sr.append(f"To: <sip:{self.user}@{self.domain}>;tag={self.tag()}")
        callid_hdr = call_id or uuid.uuid4().hex
        sr.append(f"Call-ID: {callid_hdr}")
        sr.append(f"CSeq: {cseq}")
        sr.append(f"Contact: <sip:{self.user}@{self.domain}>")
        sr.append("Content-Type: application/sdp")
        sdp_body = pc.localDescription.sdp
        sr.append(f"Content-Length: {len(sdp_body.encode('utf8'))}")
        msg = "\r\n".join(sr) + "\r\n\r\n" + sdp_body
        await self.ws.send(msg)
        print("Sent 200 OK with SDP answer")
        self.callid_to_pc[callid_hdr] = pc
        # keep pc alive in dict; ACK handled by 3CX but we don't wait specifically

    async def _consume_remote_audio(self, track, callid):
        """
        Receives audio frames from remote track. We buffer into a small file for chunk_seconds,
        then run ASR and TTS reply; when TTS is ready we push PCM to outgoing track queue.
        """
        chunk_seconds = CFG["chunk_seconds"]
        sr = CFG["sample_rate"]
        frames = []
        last_ts = time.time()
        while True:
            try:
                frame = await track.recv()
            except Exception as e:
                print("Track recv error:", e)
                break
            # convert to numpy int16
            arr = frame.to_ndarray()  # shape (channels, samples) or (samples,) depending
            # if shape is 2d, take first (mono)
            if arr.ndim > 1:
                pcm = arr[0]
            else:
                pcm = arr
            # aiortc frames are float32 in range [-1,1] typically; convert to int16
            if pcm.dtype == np.float32 or pcm.dtype == np.float64:
                pcm_i16 = (pcm * 32767).astype(np.int16)
            else:
                pcm_i16 = pcm.astype(np.int16)
            frames.append(pcm_i16)
            # if buffered enough seconds
            total_samples = sum(len(f) for f in frames)
            if total_samples >= sr * chunk_seconds:
                # write to wav
                all_pcm = np.concatenate(frames)
                ts = datetime.utcnow().strftime("%Y%m%d%H%M%S")
                wav_path = str(TMP_DIR / f"{callid}_{ts}.wav")
                sf.write(wav_path, all_pcm.astype(np.int16), sr)
                print("Wrote chunk:", wav_path)
                # spawn processing in background thread to avoid blocking recv loop
                asyncio.ensure_future(self._process_chunk_and_reply(wav_path, callid))
                frames = []

    async def _process_chunk_and_reply(self, wav_path, callid):
        # Transcribe
        try:
            text = transcribe_wav(wav_path)
            print("Transcribed:", text)
        except Exception as e:
            print("ASR error:", e)
            text = ""

        # send to LLM
        try:
            prompt = f"You are a helpful assistant for a steel manufacturer. Caller said: {text}\nRespond concisely."
            reply = query_ollama(prompt)
            print("LLM reply:", reply)
        except Exception as e:
            print("LLM error:", e)
            reply = "I am sorry, I cannot respond at the moment."

        # TTS
        tts_wav = wav_path.replace(".wav", "_tts.wav")
        try:
            generate_tts_to_wav(reply, tts_wav)
        except Exception as e:
            print("TTS generation error:", e)
            # create a short fallback voice or silence
            sf.write(tts_wav, np.zeros(int(CFG["sample_rate"] * 0.5)), CFG["sample_rate"])

        # push TTS PCM to outgoing track
        pc = None
        # find pc for this callid (callid used as key)
        for k, v in list(self.callid_to_pc.items()):
            if k and k in callid:
                pc = v
                break
        # if not matched directly, try any existing pc
        if pc is None and len(self.callid_to_pc) > 0:
            pc = list(self.callid_to_pc.values())[0]

        if pc:
            # get the outgoing track object from pc (we added one on creation)
            for sender in pc.getSenders():
                # attempt to find our OutgoingAudioTrack instance
                track = sender.track
                if isinstance(track, OutgoingAudioTrack):
                    try:
                        data, sr = sf.read(tts_wav)
                        # data returned as float or int; convert to int16 mono
                        if data.ndim > 1:
                            data = data[:,0]
                        if data.dtype == np.float32 or data.dtype == np.float64:
                            pcm = (data * 32767).astype(np.int16)
                        else:
                            pcm = data.astype(np.int16)
                        # push in chunks of 20ms
                        hop = int(sr * 0.02)
                        idx = 0
                        while idx < len(pcm):
                            chunk = pcm[idx:idx+hop]
                            # push asynchronously
                            await track.push_pcm(chunk, sr)
                            idx += hop
                            await asyncio.sleep(0)  # yield
                        print("Pushed TTS to outgoing track")
                    except Exception as e:
                        print("Error pushing TTS:", e)
                    break

        append_log({"callid": callid, "transcript": text, "reply": reply})
```

# Caveats & Practical Troubleshooting

1. **WSS Certificate / Insecure**: For production, set `insecure=False` and install 3CX TLS cert in Windows Trusted Root. Using `insecure=True` is insecure — use only for testing.
2. **SIP dialogs & header quirks**: 3CX might send slightly different SIP header formats; you may need to adapt parsing or header contents. I implemented a tolerant parser, but real PBX variety requires tweaks.
3. **ICE/STUN**: If media cannot establish (no audio), configure `CFG["stun_servers"]` (e.g., `["stun:stun.l.google.com:19302"]`) and ensure NAT/firewall allows RTP (aiortc will try ICE).
4. **ASR & TTS performance**: Whisper/Bark can be slow on CPU. Consider installing GPU, use `faster-whisper` or quantized models, or run the heavy ML stack on a separate Linux server and have this Windows agent call an internal HTTP API.
5. **3CX Version**: The code was written against typical 3CX WSS behavior. If your 3CX has call control settings or different WSS path, adapt `wss_url`.
6. **Edge cases**: Multiple simultaneous calls — this code stores `callid -> pc` mappings but may need more robust concurrency handling if you expect high call volume.
7. **DTMF / Transfers**: Not implemented here. You can detect SIP INFO or DTMF in SDP or RTP later if required.

### **2. web/app.py**

```python
from flask import Flask, render_template, jsonify, request
import csv, os, json

app = Flask(__name__)
DATA_FILE = '../logs/call_actions.csv'
responses = {}  # live override dictionary

@app.route('/')
def dashboard():
    return render_template('dashboard.html')

@app.route('/api/actions')
def actions():
    if not os.path.exists(DATA_FILE):
        return jsonify([])
    with open(DATA_FILE, 'r', encoding='utf-8') as f:
        rows = [line.strip().split(',',5) for line in f]
    data = [
        {'timestamp': r[0], 'call_id': r[1], 'story': r[2], 'intent': r[3], 'action': r[4], 'response': r[5]}
        for r in rows
    ]
    return jsonify(data)

@app.route('/api/update_response', methods=['POST'])
def update_response():
    data_req = request.get_json()
    call_id = data_req.get('call_id')
    response = data_req.get('response')
    responses[call_id] = response
    return {'status': 'ok'}

if __name__ == '__main__':
    app.run(host='0.0.0.0', port=5000)
```

```javascript
async function fetchCalls(){
    const resp = await fetch('/api/actions');
    const data = await resp.json();
    const tbody = document.querySelector('#calls-table tbody');
    tbody.innerHTML = '';
    data.forEach(call => {
        const tr = document.createElement('tr');
        tr.innerHTML = `<td>${call.timestamp}</td>
                        <td>${call.call_id}</td>
                        <td>${call.story}</td>
                        <td>${call.intent}</td>
                        <td>${call.action}</td>
                        <td><input type="text" value="${call.response}" onchange="updateResponse('${call.call_id}', this.value)"></td>`;
        tbody.appendChild(tr);
    });
}

function updateResponse(callId, newResponse){
    fetch('/api/update_response', {
        method:'POST',
        headers:{'Content-Type':'application/json'},
        body: JSON.stringify({call_id: callId, response: newResponse})
    });
}

setInterval(fetchCalls, 3000);
fetchCalls();
```
