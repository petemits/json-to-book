class EmotionAgent:

    def analyze(self,text):

        text = text.lower()

        emotions = {
            "fear":0,
            "anger":0,
            "stress":0,
            "hope":0,
            "clarity":0,
            "confidence":0
        }

        fear_words = [
            "fear",
            "afraid",
            "panic"
        ]

        for word in fear_words:
            if word in text:
                emotions["fear"] += 1

        return emotions

import sqlite3

from auth.password import (
    hash_password,
    verify_password

DATABASE = "database.db"


class AuthManager:

    def register(
        self,
        username,
        email,
        password
    ):

        conn = sqlite3.connect(
            DATABASE
        )

        cursor = conn.cursor()

        hashed = hash_password(
            password
        )

        cursor.execute(
            """
            INSERT INTO users
            (
                username,
                email,
                password
            )
            VALUES(?,?,?)
            """,
            (
                username,
                email,
                hashed
            )
        )

        conn.commit()

        conn.close()

        return True

    def login(
        self,
        email,
        password
    ):

        conn = sqlite3.connect(
            DATABASE
        )

        cursor = conn.cursor()

        cursor.execute(
            """
            SELECT *
            FROM users
            WHERE email=?
            """,
            (email,)
        )

        user = cursor.fetchone()

        conn.close()

        if not user:
            return None

        if verify_password(
            password,
            user[3]
        ):
            return user

        return None
