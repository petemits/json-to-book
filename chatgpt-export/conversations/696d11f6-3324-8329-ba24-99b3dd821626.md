```python
import time
import os
import pandas as pd
from selenium import webdriver
from selenium.webdriver.common.by import By
from selenium.webdriver.common.keys import Keys
from selenium.webdriver.firefox.options import Options

# ================= USER SETTINGS =================

DOWNLOAD_FILES = True        # Set False if you don't want downloads
RESULT_PAGES = 2             # 1 or 2 pages
WAIT_TIME = 3                # Seconds between actions
OUTPUT_FILE = "scraped_data.xlsx"
DOWNLOAD_DIR = os.path.abspath("downloads")

# =================================================

def setup_firefox():
    options = Options()
    options.set_preference("browser.download.folderList", 2)
    options.set_preference("browser.download.dir", DOWNLOAD_DIR)
    options.set_preference("browser.helperApps.neverAsk.saveToDisk",
                            "application/pdf,image/jpeg,image/png,application/octet-stream")
    options.set_preference("pdfjs.disabled", True)

    driver = webdriver.Firefox(options=options)
    driver.maximize_window()
    return driver


def get_user_input():
    print("\nSearch Options:")
    print("1. Word")
    print("2. Sentence")
    print("3. URL")
    choice = input("Choose (1/2/3): ").strip()

    if choice not in ["1", "2", "3"]:
        raise ValueError("Invalid option")

    query = input("Enter search text or URL: ").strip()
    return query


def search_and_scrape(driver, query):
    data = []

    driver.get("https://duckduckgo.com/")
    time.sleep(WAIT_TIME)

    search_box = driver.find_element(By.NAME, "q")
    search_box.send_keys(query)
    search_box.send_keys(Keys.RETURN)
    time.sleep(WAIT_TIME)

    for page in range(RESULT_PAGES):
        results = driver.find_elements(By.CSS_SELECTOR, "article")

        for r in results:
            try:
                title = r.find_element(By.TAG_NAME, "h2").text
                link = r.find_element(By.TAG_NAME, "a").get_attribute("href")
                snippet = r.text

                print(f"Captured: {title}")

                if DOWNLOAD_FILES:
                    try:
                        driver.execute_script("window.open(arguments[0]);", link)
                        driver.switch_to.window(driver.window_handles[-1])
                        time.sleep(WAIT_TIME)
                        driver.close()
                        driver.switch_to.window(driver.window_handles[0])
                    except:
                        pass

                data.append({
                    "Title": title,
                    "URL": link,
                    "Snippet": snippet
                })

            except:
                continue

        # Go to next page
        try:
            next_btn = driver.find_element(By.LINK_TEXT, "Next")
            next_btn.click()
            time.sleep(WAIT_TIME)
        except:
            break

    return data


def save_to_excel(data):
    df = pd.DataFrame(data)
    df.to_excel(OUTPUT_FILE, index=False)
    print(f"\nSaved {len(df)} records to {OUTPUT_FILE}")


def main():
    os.makedirs(DOWNLOAD_DIR, exist_ok=True)

    query = get_user_input()
    driver = setup_firefox()

    try:
        scraped_data = search_and_scrape(driver, query)
        save_to_excel(scraped_data)
    finally:
        driver.quit()


if __name__ == "__main__":
    main()
```

```python
import time
import os
import random
import pandas as pd
from selenium import webdriver
from selenium.webdriver.common.by import By
from selenium.webdriver.common.keys import Keys
from selenium.webdriver.firefox.options import Options

# ================= USER SETTINGS =================

MAX_PAGES = 2                    # Search up to page 2
MAX_RANDOM_RESULTS = 10          # Random number of links per run
MIN_DELAY = 2                    # Seconds
MAX_DELAY = 6                    # Seconds
DOWNLOAD_FILES = False           # Set True to open/download files
OUTPUT_FILE = "scraped_data.xlsx"
DOWNLOAD_DIR = os.path.abspath("downloads")

# =================================================


def random_sleep():
    time.sleep(random.uniform(MIN_DELAY, MAX_DELAY))


def setup_firefox():
    options = Options()
    options.set_preference("browser.download.folderList", 2)
    options.set_preference("browser.download.dir", DOWNLOAD_DIR)
    options.set_preference("browser.helperApps.neverAsk.saveToDisk",
                            "application/pdf,image/jpeg,image/png,application/octet-stream")
    options.set_preference("pdfjs.disabled", True)

    driver = webdriver.Firefox(options=options)
    driver.maximize_window()

def get_user_input():
    print("\nSearch Options:")
    print("1. Word")
    print("2. Sentence")
    print("3. URL")
    choice = input("Choose (1/2/3): ").strip()

    if choice not in ["1", "2", "3"]:
        raise ValueError("Invalid option")

    query = input("Enter search text or URL: ").strip()
    return query


def search_and_scrape(driver, query):
    data = []

    driver.get("https://duckduckgo.com/")
    random_sleep()

    search_box = driver.find_element(By.NAME, "q")
    search_box.send_keys(query)
    search_box.send_keys(Keys.RETURN)
    random_sleep()

    collected_links = []

    for page in range(MAX_PAGES):
        results = driver.find_elements(By.CSS_SELECTOR, "article")

        for r in results:
            try:
                title = r.find_element(By.TAG_NAME, "h2").text
                link = r.find_element(By.TAG_NAME, "a").get_attribute("href")
                snippet = r.text

                collected_links.append({
                    "Title": title,
                    "URL": link,
                    "Snippet": snippet
                })
            except:
                continue

        # Move to next page randomly
        try:
            if random.choice([True, False]):
                next_btn = driver.find_element(By.LINK_TEXT, "Next")
                next_btn.click()
                random_sleep()
            else:
                break
        except:
            break

    # 🔀 RANDOMIZE RESULTS
    random.shuffle(collected_links)

    # 🎯 PICK RANDOM SUBSET
    selected = random.sample(
        collected_links,
        min(len(collected_links), random.randint(3, MAX_RANDOM_RESULTS))
    )

    # 🔗 OPTIONALLY OPEN RANDOM LINKS
    for item in selected:
        print(f"Captured: {item['Title']}")

        if DOWNLOAD_FILES:
            try:
                driver.execute_script("window.open(arguments[0]);", item["URL"])
                driver.switch_to.window(driver.window_handles[-1])
                random_sleep()
                driver.close()
                driver.switch_to.window(driver.window_handles[0])
            except:
                pass

        data.append(item)

    return data


def save_to_excel(data):
    df = pd.DataFrame(data)
    df.to_excel(OUTPUT_FILE, index=False)
    print(f"\nSaved {len(df)} randomized records to {OUTPUT_FILE}")

    query = get_user_input()
    driver = setup_firefox()

        scraped_data = search_and_scrape(driver, query)
```

```python
from sklearn.metrics.pairwise import cosine_similarity


# ================= LOAD CONFIG =================

with open("config.json") as f:
    CONFIG = json.load(f)

# ===============================================


# ================= VOICE ENGINE =================

engine = pyttsx3.init()
engine.setProperty("rate", 165)

def speak(text):
    print("AI:", text)
    engine.say(text)
    engine.runAndWait()

# ===============================================


# ================= SPEECH INPUT =================

def listen():
    r = sr.Recognizer()
    with sr.Microphone() as source:
        speak("I am listening.")
        audio = r.listen(source)

    try:
        text = r.recognize_google(audio)
        print("You:", text)
        return text
    except:
        speak("I did not understand.")
        return None

# ================= SEMANTIC NLP =================

INTENTS = [
    "search the web",
    "find information",
    "look up a topic",
    "scrape data",
    "collect results",
    "download information",
    "exit program"
]

def detect_intent(user_text):
    vect = TfidfVectorizer()
    X = vect.fit_transform(INTENTS + [user_text])
    scores = cosine_similarity(X[-1], X[:-1])
    return INTENTS[scores.argmax()]

# ===============================================

    driver = webdriver.Firefox(options=options)
    driver.maximize_window()
```

```python
# neural_firefox_assistant.py
# Python 3.8 compatible
```

import os
import time
import random
from datetime import datetime

import pandas as pd
import speech_recognition as sr
import pyttsx3

from selenium import webdriver
from selenium.webdriver.common.by import By
from selenium.webdriver.common.keys import Keys
from selenium.webdriver.firefox.options import Options

from sentence_transformers import SentenceTransformer
from newspaper import Article

from flask import Flask, request

# =========================
# CONFIG
# =========================
BASE_DIR = os.getcwd()
DOWNLOAD_DIR = os.path.join(BASE_DIR, "downloads")
os.makedirs(DOWNLOAD_DIR, exist_ok=True)

EXCEL_FILE = os.path.join(
    BASE_DIR,
    "scraped_" + datetime.now().strftime("%Y%m%d_%H%M%S") + ".xlsx"
)

MAX_PAGES = 2
MIN_DELAY = 2
MAX_DELAY = 4

# =========================
# INIT ENGINES
# =========================
engine = pyttsx3.init()
engine.setProperty("rate", 155)

model = SentenceTransformer("all-MiniLM-L6-v2")

# =========================
# VOICE
# =========================
def speak(text):
    print("AI:", text)
    engine.say(text)
    engine.runAndWait()

def listen():
    r = sr.Recognizer()
    with sr.Microphone() as source:
        r.adjust_for_ambient_noise(source)
        audio = r.listen(source)
    try:
        return r.recognize_google(audio)
    except:
        return None

# =========================
# INPUT MODE
# =========================
def get_user_input():
    print("\nChoose input:")
    print("1. Type")
    print("2. Speak")
    print("3. Exit")

    choice = input("Select: ").strip()

    if choice == "1":
        return input("YOU: ")
    elif choice == "2":
        speak("I am listening")
        return listen()
    elif choice == "3":
        return "exit"
    return None

# =========================
# FIREFOX SETUP
# =========================
def setup_firefox():
    options = Options()
    options.set_preference("browser.download.folderList", 2)
    options.set_preference("browser.download.dir", DOWNLOAD_DIR)
    options.set_preference("pdfjs.disabled", True)
    return webdriver.Firefox(options=options)

# =========================
# SEARCH + SCRAPE
# =========================
def search_and_scrape(topic):
    driver = setup_firefox()
    collected = []

    driver.get("https://duckduckgo.com/")
    time.sleep(2)

    box = driver.find_element(By.NAME, "q")
    box.send_keys(topic)
    box.send_keys(Keys.RETURN)
    time.sleep(3)

    for _ in range(MAX_PAGES):
        results = driver.find_elements(By.CSS_SELECTOR, "article")

        for r in results:
            try:
                link = r.find_element(By.TAG_NAME, "a").get_attribute("href")
                article = Article(link)
                article.download()
                article.parse()

                collected.append({
                    "Topic": topic,
                    "Title": article.title,
                    "URL": link,
                    "Text": article.text[:2000]
                })
            except:
                continue

        try:
            driver.find_element(By.LINK_TEXT, "More Results").click()
            time.sleep(random.uniform(MIN_DELAY, MAX_DELAY))
        except:
            break

    driver.quit()
    return collected

# =========================
# SAVE TO EXCEL
# =========================
def save_to_excel(rows):
    if not rows:
        return
    df = pd.DataFrame(rows)
    df.to_excel(EXCEL_FILE, index=False)

# =========================
# SEMANTIC TOPIC SHIFT
# =========================
def related_topic(topic):
    modifiers = [
        "future of",
        "benefits of",
        "risks of",
        "applications of",
        "comparison of",
        "challenges in"
    ]
    return random.choice(modifiers) + " " + topic

# =========================
# TERMINAL MODE

def terminal_assistant():
    speak("Neural Firefox assistant ready.")

    current_topic = None

    while True:

        speak("Searching " + current_topic)

@app.route("/", methods=["GET", "POST"])

        save_to_excel(results)
        return "<h2>Scraped {} articles</h2>".format(len(results))

    return """
    <html>
    <body>
      <h2>Neural Firefox Assistant</h2>
      <form method="post">
        <input name="topic" placeholder="Enter topic" />
        <button type="submit">Search</button>
      </form>
    </body>

    print("1. Terminal Assistant")

    mode = input("Select: ").strip()

    if mode == "2":
