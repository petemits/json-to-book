# Python script
Save as `generate_random_video.py`. This script:

- Sends prompt to Hugging Face inference endpoint for a text-to-video model
- Saves returned MP4 (handles both raw video bytes and base64-in-json responses)

```python
#!/usr/bin/env python3
"""
generate_random_video.py
Requires: pip install requests python-dotenv
Set env vars: OPENROUTER_API_KEY, HF_API_TOKEN
"""

import os
import time
import requests
import base64
import json
from datetime import datetime
from dotenv import load_dotenv
import random

load_dotenv()  # optional .env support

OPENROUTER_API_KEY = os.getenv("OPENROUTER_API_KEY")
HF_API_TOKEN = os.getenv("HF_API_TOKEN")

if not OPENROUTER_API_KEY or not HF_API_TOKEN:
    raise SystemExit("Missing OPENROUTER_API_KEY or HF_API_TOKEN environment variables.")

# ---------- Configuration ----------
# Choose a Hugging Face hosted model id that supports text-to-video.
# You may change this to any model id on Hugging Face that supports the text-to-video task.
HF_MODEL = "runwayml/stable-diffusion-v1-5"  # example; replace with actual text-to-video model if available

# OpenRouter chat model (you can use the default endpoint)
OPENROUTER_URL = "https://openrouter.ai/v1/chat/completions"

# Hugging Face inference endpoint:
HF_INFERENCE_URL = f"https://api-inference.huggingface.co/models/{HF_MODEL}"

# ---------- Helper functions ----------
def ask_openrouter_for_prompt(topic="Toronto GTA condo prices analyses"):
    # Compose a system+user instruction to create a random, creative short video prompt
    system = "You are a concise creative assistant that outputs one short video prompt for a text-to-video model. Use 1-2 short sentences and include visuals, style, and duration."
    user = (
        f"Give me a single, random video prompt (one or two sentences) about '{topic}'. "
        "Include a suggested duration (~6-20s) and a short style tag (e.g., 'animated infographic', 'cinematic drone', '2D explainer'). "
        "Make it unique each time and suitable for automatic text-to-video generation."
    )

    headers = {
        "Authorization": f"Bearer {OPENROUTER_API_KEY}",
        "Content-Type": "application/json",
    }
    payload = {
        "model": "gpt-4o-mini",   # OpenRouter accepts many model names; adjust if yours uses another name
        "messages": [
            {"role": "system", "content": system},
            {"role": "user", "content": user}
        ],
        "temperature": 0.9,
        "max_tokens": 200
    }

    resp = requests.post(OPENROUTER_URL, headers=headers, json=payload, timeout=120)
    resp.raise_for_status()
    resj = resp.json()

    # OpenRouter response shape is similar to OpenAI Chat: extract text
    try:
        text = resj["choices"][0]["message"]["content"].strip()
    except Exception:
        # best-effort fallback
        text = json.dumps(resj)[:400]
    return text

def call_hf_text_to_video(prompt):
    """
    Calls the Hugging Face inference API for the specified model.
    Many HF video endpoints accept JSON: {"inputs": "<prompt>", "parameters": {...}}
    The endpoint can return video bytes directly or a JSON with base64; this function handles both.
    """
    headers = {"Authorization": f"Bearer {HF_API_TOKEN}"}
    payload = {"inputs": prompt, "parameters": {"max_length": 64}}

    print("Sending prompt to Hugging Face inference endpoint...")
    r = requests.post(HF_INFERENCE_URL, headers=headers, json=payload, stream=True, timeout=600)

    # If Hugging Face returns JSON with base64 video:
    content_type = r.headers.get("Content-Type", "")
    if "application/json" in content_type:
        j = r.json()
        # common pattern: output might have 'generated_video' or 'video' fields; try to find base64
        if isinstance(j, dict):
            for k in ("generated_video", "video", "data", "output"):
                if k in j:
                    val = j[k]
                    if isinstance(val, str):
                        # assume base64
                        try:
                            return base64.b64decode(val)
                        except Exception:
                            pass
                    elif isinstance(val, list) and val and isinstance(val[0], str):
                        # maybe data:[ "data:video/mp4;base64,...." ]
                        s = val[0]
                        if "," in s:
                            b64 = s.split(",", 1)[1]
                            return base64.b64decode(b64)
            # fallback: scan the json for any base64-looking string
            def find_b64(obj):
                if isinstance(obj, str) and len(obj) > 100 and all(c.isalnum() or c in "+/=:\n\r" for c in obj[:50]):
                    return obj
                if isinstance(obj, dict):
                    for v in obj.values():
                        res = find_b64(v)
                        if res:
                            return res
                if isinstance(obj, list):
                    for v in obj:
                        res = find_b64(v)
                        if res:
                            return res
                return None
            candidate = find_b64(j)
            if candidate:
                # strip data:...prefix if present
                if "," in candidate:
                    candidate = candidate.split(",",1)[1]
                return base64.b64decode(candidate)
        raise RuntimeError("HF inference returned JSON but no base64 video field found: " + json.dumps(j)[:400])

    # If the endpoint returns binary video bytes (content-type video/mp4)
    if "video" in content_type or "octet-stream" in content_type:
        return r.content

    # Unexpected response
    raise RuntimeError(f"Unexpected response content-type from HF: {content_type}")

def save_video_bytes(b, prompt):
    ts = datetime.now().strftime("%Y%m%d_%H%M%S")
    rand = random.randint(1000,9999)
    safe = "".join([c if c.isalnum() else "_" for c in prompt])[:60]
    filename = f"toronto_condo_analysis_{ts}_{rand}.mp4"
    outdir = os.path.join(os.getcwd(), "generated_videos")
    os.makedirs(outdir, exist_ok=True)
    path = os.path.join(outdir, filename)
    with open(path, "wb") as f:
        f.write(b)
    return path

# ---------- Main ----------
def main():
    print("Asking OpenRouter for a random video prompt...")
    prompt = ask_openrouter_for_prompt()
    print("Prompt received:")
    print(prompt)
    # Optionally wrap prompt with a small instruction for the HF model:
    hf_prompt = f"{prompt}\n\nMake a short stylized video roughly 8-12 seconds, focusing on visuals and animated infographic elements about Toronto GTA condo prices. Output must be a short video with clear visuals."

    try:
        video_bytes = call_hf_text_to_video(hf_prompt)
    except Exception as e:
        print("Error calling Hugging Face inference API:", e)
        return

    path = save_video_bytes(video_bytes, prompt)
    print("Saved generated video to:", path)

if __name__ == "__main__":
    main()
```

# Java script (single-file)
This Java program (requires Java 11+) does the same: asks OpenRouter, then calls Hugging Face inference endpoint and saves MP4. Save as `GenerateRandomVideo.java`.

```java
// Compile: javac GenerateRandomVideo.java
// Run: java -DOPENROUTER_API_KEY=... -DHF_API_TOKEN=... GenerateRandomVideo

import java.net.http.*;
import java.net.*;
import java.io.*;
import java.nio.file.*;
import java.time.*;
import java.util.*;
import com.fasterxml.jackson.databind.*; // optional - but below we avoid external deps and do naive parsing

public class GenerateRandomVideo {
    public static void main(String[] args) throws Exception {
        String openrouterKey = System.getProperty("OPENROUTER_API_KEY");
        String hfKey = System.getProperty("HF_API_TOKEN");
        if (openrouterKey == null || hfKey == null) {
            System.err.println("Run with -DOPENROUTER_API_KEY=... -DHF_API_TOKEN=...");
            return;
        }

        String openrouterUrl = "https://openrouter.ai/v1/chat/completions";
        String hfModel = "runwayml/stable-diffusion-v1-5"; // replace with a text-to-video model id if needed
        String hfUrl = "https://api-inference.huggingface.co/models/" + hfModel;

        String prompt = askOpenRouterForPrompt(openrouterUrl, openrouterKey);
        System.out.println("Prompt from OpenRouter:\n" + prompt);

        String hfPrompt = prompt + "\n\nMake an animated short (8-12s) visualizing Toronto GTA condo prices with simple moving charts and skyline shots.";
        byte[] videoBytes = callHfInference(hfUrl, hfKey, hfPrompt);

        if (videoBytes != null) {
            Path outDir = Paths.get("generated_videos");
            Files.createDirectories(outDir);
            String ts = DateTimeFormatter.ofPattern("yyyyMMdd_HHmmss").format(LocalDateTime.now());
            String filename = String.format("toronto_condo_analysis_%s_%d.mp4", ts, new Random().nextInt(9000)+1000);
            Path outPath = outDir.resolve(filename);
            Files.write(outPath, videoBytes);
            System.out.println("Saved video to: " + outPath.toAbsolutePath().toString());
        } else {
            System.err.println("No video bytes returned.");
        }
    }

    static String askOpenRouterForPrompt(String url, String apiKey) throws Exception {
        String system = "You are a concise creative assistant that outputs one short video prompt for a text-to-video model.";
        String user = "Give me a single, random video prompt about 'Toronto GTA condo prices analyses'. Include duration (~6-20s) and style tag. Make it unique.";

        String payload = "{"
                + "\"model\":\"gpt-4o-mini\","
                + "\"messages\":[{\"role\":\"system\",\"content\":\"" + escapeJson(system) + "\"},{\"role\":\"user\",\"content\":\"" + escapeJson(user) + "\"}],"
                + "\"temperature\":0.9"
                + "}";

        HttpClient client = HttpClient.newHttpClient();
        HttpRequest req = HttpRequest.newBuilder()
                .uri(URI.create(url))
                .header("Authorization", "Bearer " + apiKey)
                .header("Content-Type", "application/json")
                .POST(HttpRequest.BodyPublishers.ofString(payload))
                .build();
        HttpResponse<String> resp = client.send(req, HttpResponse.BodyHandlers.ofString());
        if (resp.statusCode() >= 400) throw new IOException("OpenRouter error: " + resp.body());

        // naive extraction of choices[0].message.content
        String body = resp.body();
        int idx = body.indexOf("\"content\"");
        if (idx >= 0) {
            int colon = body.indexOf(":", idx);
            int firstQuote = body.indexOf("\"", colon+1);
            int secondQuote = body.indexOf("\"", firstQuote+1);
            if (firstQuote >=0 && secondQuote>firstQuote) {
                return body.substring(firstQuote+1, secondQuote).replace("\\n", "\n");
            }
        }
        return body.length() > 300 ? body.substring(0,300) : body;
    }

    static byte[] callHfInference(String url, String apiKey, String prompt) throws Exception {
        HttpClient client = HttpClient.newBuilder().build();
        String payload = "{\"inputs\": " + JSONObjectEscape(prompt) + "}";
        HttpRequest req = HttpRequest.newBuilder()
                .uri(URI.create(url))
                .header("Authorization", "Bearer " + apiKey)
                .header("Content-Type", "application/json")
                .POST(HttpRequest.BodyPublishers.ofString(payload))
                .build();

        HttpResponse<byte[]> response = client.send(req, HttpResponse.BodyHandlers.ofByteArray());
        String contentType = response.headers().firstValue("Content-Type").orElse("");
        if (contentType.contains("application/json")) {
            String txt = new String(response.body());
            // Try to find base64 chunk (very naive search for "data:video" or a long base64)
            int idx = txt.indexOf("base64,");
            if (idx>=0) {
                String base64 = txt.substring(idx+7).replaceAll("[^A-Za-z0-9+/=]", "");
                return Base64.getDecoder().decode(base64);
            } else {
                // try to find a long string
                int start = txt.indexOf("\"");
                if (start>=0) {
                    String candidate = txt.substring(start+1);
                    if (candidate.length() > 200) {
                        candidate = candidate.split("\"")[0];
                        return Base64.getDecoder().decode(candidate);
                    }
                }
            }
            throw new IOException("HF returned JSON but no base64 video found.");
        } else if (contentType.contains("video") || contentType.contains("octet-stream")) {
            return response.body();
        } else {
            throw new IOException("Unexpected HF content-type: " + contentType + " body len=" + response.body().length);
        }
    }

    static String escapeJson(String s) {
        return s.replace("\\", "\\\\").replace("\"", "\\\"").replace("\n", "\\n");
    }

    static String JSONObjectEscape(String s) {
        return "\"" + s.replace("\\", "\\\\").replace("\"", "\\\"").replace("\n", "\\n") + "\"";
    }
}
```

```python
#!/usr/bin/env python3
"""
Random AI Text-to-Video Generator
---------------------------------
This script:
1. Gets a random creative prompt from OpenRouter
2. Sends it to a Hugging Face text-to-video model
3. Saves the resulting video file locally

Requirements:
    pip install requests python-dotenv
"""

import os
import time
import json
import base64
import random
import requests
from datetime import datetime
from dotenv import load_dotenv

# Load .env file (optional)
load_dotenv()

# ---------- Load JSON Config ----------
CONFIG_FILE = "config.json"
if not os.path.exists(CONFIG_FILE):
    raise SystemExit(f"Missing {CONFIG_FILE}. Create one (see example).")

with open(CONFIG_FILE, "r") as f:
    config = json.load(f)

# ---------- Environment Variables ----------
OPENROUTER_API_KEY = os.getenv("OPENROUTER_API_KEY") or config.get("OPENROUTER_API_KEY")
HF_API_TOKEN = os.getenv("HF_API_TOKEN") or config.get("HF_API_TOKEN")

if not OPENROUTER_API_KEY or not HF_API_TOKEN:
    raise SystemExit("Missing OPENROUTER_API_KEY or HF_API_TOKEN. Add them to .env or config.json.")

# ---------- API Endpoints ----------
OPENROUTER_URL = "https://openrouter.ai/v1/chat/completions"
HF_MODEL = config["HUGGINGFACE_MODEL"]
HF_URL = f"https://api-inference.huggingface.co/models/{HF_MODEL}"

# ---------- Functions ----------

def get_random_prompt(topic):
    """Ask OpenRouter to generate a random video prompt."""
    headers = {
        "Authorization": f"Bearer {OPENROUTER_API_KEY}",
        "Content-Type": "application/json",
    }

    payload = {
        "model": config["OPENROUTER_MODEL"],
        "temperature": config.get("temperature", 0.9),
        "messages": [
            {"role": "system", "content": "You are a creative assistant that writes short video ideas."},
            {"role": "user", "content": f"Give me one random video prompt about '{topic}'. "
                                        "Use 1-2 sentences. Mention visuals, tone, and duration (~10s)."}
        ],
    }

    print("🧠 Requesting random idea from OpenRouter...")
    r = requests.post(OPENROUTER_URL, headers=headers, json=payload, timeout=120)
    r.raise_for_status()
    data = r.json()
    prompt = data["choices"][0]["message"]["content"].strip()
    print("💡 Prompt generated:", prompt)
    return prompt


def generate_video_from_prompt(prompt):
    """Call Hugging Face API to generate a short video."""
    headers = {"Authorization": f"Bearer {HF_API_TOKEN}"}
    payload = {"inputs": prompt}

    print("🎬 Generating video via Hugging Face...")
    response = requests.post(HF_URL, headers=headers, json=payload, stream=True, timeout=600)
    content_type = response.headers.get("Content-Type", "")

    if "application/json" in content_type:
        j = response.json()
        if "error" in j:
            raise RuntimeError(f"Hugging Face error: {j['error']}")
        # Handle base64 video
        if isinstance(j, dict):
            for k in j:
                if isinstance(j[k], str) and j[k].startswith("data:video"):
                    base64_data = j[k].split(",", 1)[1]
                    return base64.b64decode(base64_data)
        raise RuntimeError("No video data found in JSON response.")
    elif "video" in content_type or "octet-stream" in content_type:
        return response.content
    else:
        raise RuntimeError(f"Unexpected content type: {content_type}")


def save_video(video_bytes, prompt):
    """Save generated video file locally."""
    os.makedirs("videos", exist_ok=True)
    timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
    random_id = random.randint(1000, 9999)
    filename = f"toronto_condo_{timestamp}_{random_id}.mp4"
    filepath = os.path.join("videos", filename)
    with open(filepath, "wb") as f:
        f.write(video_bytes)
    print(f"✅ Video saved as {filepath}")
    return filepath


# ---------- Main ----------
def main():
    topic = config.get("topic", "Toronto GTA condo prices analysis")

    try:
        prompt = get_random_prompt(topic)
        video_bytes = generate_video_from_prompt(prompt)
        save_video(video_bytes, prompt)
    except Exception as e:
        print("❌ Error:", e)


if __name__ == "__main__":
    main()
```

```bash
"""
Random AI Text-to-Video Generator
---------------------------------
Requirements:
    pip install requests python-dotenv
Set API keys either in a .env file or in config.json

import os
import json
import base64
import random

from datetime import datetime
from dotenv import load_dotenv

CONFIG_FILE = "config.json"
if not os.path.exists(CONFIG_FILE):

OPENROUTER_API_KEY = os.getenv("OPENROUTER_API_KEY") or config.get("OPENROUTER_API_KEY")
HF_API_TOKEN = os.getenv("HF_API_TOKEN") or config.get("HF_API_TOKEN")

if not OPENROUTER_API_KEY or not HF_API_TOKEN:
    raise SystemExit("Missing OPENROUTER_API_KEY or HF_API_TOKEN. Set them in .env or config.json")

OPENROUTER_URL = "https://openrouter.ai/v1/chat/completions"
HF_MODEL = config.get("HUGGINGFACE_MODEL", "runwayml/stable-diffusion-v1-5")
HF_URL = f"https://api-inference.huggingface.co/models/{HF_MODEL}"

def get_random_prompt(topic):
    headers = {
        "Authorization": f"Bearer {OPENROUTER_API_KEY}",
        "Content-Type": "application/json",
    }
    payload = {
        "model": config.get("OPENROUTER_MODEL", "gpt-4o-mini"),
        "temperature": config.get("temperature", 0.9),
        "messages": [
            {"role": "system", "content": "You are a creative assistant that writes short video ideas."},
            {"role": "user", "content": f"Give me one random video prompt about '{topic}'. Use 1-2 sentences. Mention visuals, tone, and duration (~10s)."}
        ],
    }
    r = requests.post(OPENROUTER_URL, headers=headers, json=payload, timeout=120)
    r.raise_for_status()
    j = r.json()
    prompt = j["choices"][0]["message"]["content"].strip()
    return prompt

def generate_video_from_prompt(prompt):
    headers = {"Authorization": f"Bearer {HF_API_TOKEN}"}
    payload = {"inputs": prompt}
    r = requests.post(HF_URL, headers=headers, json=payload, stream=True, timeout=600)
    content_type = r.headers.get("Content-Type", "")
    if "application/json" in content_type:
        j = r.json()
        if "error" in j:
            raise RuntimeError(f"Hugging Face error: {j['error']}")
        # look for base64 data fields
        if isinstance(j, dict):
            # common field patterns
            for k in ("generated_video", "video", "data", "output"):
                val = j.get(k)
                if isinstance(val, str) and val.startswith("data:video"):
                    b64 = val.split(",",1)[1]
                    return base64.b64decode(b64)
            # fallback: scan for large string
            def find_b64(obj):
                if isinstance(obj, str) and len(obj) > 200 and ("base64" in obj or all(c.isalnum() or c in "+/=:" for c in obj[:50])):
                    return obj
                if isinstance(obj, dict):
                    for v in obj.values():
                        r = find_b64(v)
                        if r:
                            return r
                if isinstance(obj, list):
                    for v in obj:
                        r = find_b64(v)
                        if r:
                            return r
                return None
            cand = find_b64(j)
            if cand:
                if "," in cand:
                    cand = cand.split(",",1)[1]
                return base64.b64decode(cand)
        raise RuntimeError("No video data found in HF JSON response.")
    elif "video" in content_type or "octet-stream" in content_type:
        return r.content
    else:
        raise RuntimeError(f"Unexpected HF content-type: {content_type}")

def save_video(video_bytes, prompt):
```

```bash
from datetime import datetime
```

```python
OPENROUTER_API_KEY = os.getenv("OPENROUTER_API_KEY")

# List of fallback prompts in case OpenRouter fails
fallback_prompts = [

    "GTA real estate trends in 2025",
    "Investing in Toronto condos explained",
    "Future of real estate in Toronto GTA",

def get_prompt(topic):
    """Try to get a creative prompt from OpenRouter. Fallback if it fails."""
    url = "https://openrouter.ai/v1/chat/completions"
    headers = {
        "Authorization": f"Bearer {OPENROUTER_API_KEY}",

        ]

    try:
        r = requests.post(url, headers=headers, json=data, timeout=10)

        filename = f"videos/video_{datetime.now().strftime('%Y%m%d_%H%M%S')}_{random.randint(1000,9999)}.mp4"
```

```python
import requests
import os
import random
from datetime import datetime

OUTPUT_FOLDER = "generated_images"
os.makedirs(OUTPUT_FOLDER, exist_ok=True)

TOPICS = [
    "Affordable condos in downtown Toronto",
    "Luxury homes near the CN Tower",
    "Future smart housing developments in GTA",
    "Urban skyline and condo investments in Mississauga",
    "Sustainable living apartments in Toronto"
]

def fetch_image_from_lexica(prompt):
    print(f"🎨 Searching AI image for: {prompt}")
    url = f"https://lexica.art/api/v1/search?q={prompt}"

    response = requests.get(url)
    if response.status_code == 200:
        data = response.json()
        if data.get("images"):
            image_url = random.choice(data["images"])["src"]
            image_data = requests.get(image_url).content
            filename = os.path.join(
                OUTPUT_FOLDER,
                f"{prompt[:40].replace(' ', '_')}_{datetime.now().strftime('%Y%m%d_%H%M%S')}.jpg"
            )
            with open(filename, "wb") as f:
                f.write(image_data)
            print(f"✅ Image saved: {filename}")
        else:
            print("⚠️ No images found for this topic.")
    else:
        print(f"❌ API request failed ({response.status_code}): {response.text}")

def main():
    topic = random.choice(TOPICS)
    prompt = f"Create a detailed, photorealistic image of {topic}"
    fetch_image_from_lexica(prompt)

if __name__ == "__main__":
    main()
```
